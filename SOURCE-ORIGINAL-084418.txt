1. DATABASE SCHEMA (Supabase PostgreSQL)
sql
-- ============================================================
-- TVET e-ASSESS — COMPLETE DATABASE SCHEMA
-- Supabase PostgreSQL

-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ============================================================
-- 1. INSTITUTIONS
-- ============================================================
CREATE TABLE institutions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    address TEXT,
    phone VARCHAR(50),
    email VARCHAR(100),
    logo_url TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================
-- 2. DEPARTMENTS
-- ============================================================
CREATE TABLE departments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL,
    hod_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(institution_id, code)
);

-- ============================================================
-- 3. COURSES
-- ============================================================
CREATE TABLE courses (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL,
    qualification_level VARCHAR(50), -- e.g., Level 4, 5, 6
    duration_months INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(department_id, code)
);

-- ============================================================
-- 4. CLASSES
-- ============================================================
CREATE TABLE classes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL, -- e.g., "EEN/001/2027"
    intake_year INTEGER,
    cycle INTEGER, -- 1, 2, 3, 4, 5
    level INTEGER, -- 4, 5, 6
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, name)
);

-- ============================================================
-- 5. UNITS
-- ============================================================
CREATE TABLE units (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    code VARCHAR(50) NOT NULL, -- e.g., ENG/CU/EICR/03/5/MA
    name VARCHAR(255) NOT NULL, -- e.g., "Install Security System"
    credit_hours INTEGER,
    level INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, code)
);

-- ============================================================
-- 6. USERS (extends Supabase auth.users)
-- ============================================================
CREATE TABLE users (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    institution_id UUID REFERENCES institutions(id),
    department_id UUID REFERENCES departments(id),
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(50),
    staff_number VARCHAR(50),
    role VARCHAR(50) NOT NULL CHECK (role IN (
        'super_admin', 'hod', 'trainer', 'assessor',
        'internal_verifier', 'external_verifier', 'trainee', 'management'
    )),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================
-- 7. TRAINEES
-- ============================================================
CREATE TABLE trainees (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    admission_number VARCHAR(50) UNIQUE NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    national_id VARCHAR(50),
    phone VARCHAR(50),
    email VARCHAR(255),
    enrollment_date DATE,
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'graduated', 'dropped', 'transferred')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================
-- 8. TRAINER ALLOCATIONS
-- ============================================================
CREATE TABLE trainer_allocations (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    trainer_id UUID REFERENCES users(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    unit_id UUID REFERENCES units(id) ON DELETE CASCADE,
    academic_year INTEGER NOT NULL,
    term INTEGER NOT NULL CHECK (term IN (1, 2, 3)),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(trainer_id, class_id, unit_id, academic_year, term)
);

-- ============================================================
-- 9. ACADEMIC YEARS & TERMS
-- ============================================================
CREATE TABLE academic_periods (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    year INTEGER NOT NULL,
    term INTEGER NOT NULL CHECK (term IN (1, 2, 3)),
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(institution_id, year, term)
);

-- ============================================================
-- 10. ASSESSMENTS (Instruments)
-- ============================================================
CREATE TABLE assessments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    unit_id UUID REFERENCES units(id) ON DELETE CASCADE,
    academic_period_id UUID REFERENCES academic_periods(id),

    title VARCHAR(255) NOT NULL,
    assessment_number VARCHAR(50) NOT NULL, -- e.g., "AS01", "PA1"
    assessment_type VARCHAR(20) NOT NULL CHECK (assessment_type IN ('oral', 'practical')),
    assessment_level VARCHAR(20), -- e.g., "Level 4,5 & 6"

    instructions TEXT,
    duration_minutes INTEGER,
    total_marks INTEGER,
    passing_percentage DECIMAL(5,2) DEFAULT 50.00,

    required_tools TEXT,
    required_equipment TEXT,
    required_materials TEXT,
    safety_precautions TEXT,
    ppe_requirements TEXT,

    marking_guidelines TEXT,
    reference_documents TEXT,

    status VARCHAR(20) DEFAULT 'draft' CHECK (status IN (
        'draft', 'preview', 'published', 'withdrawn', 'archived'
    )),
    version INTEGER DEFAULT 1,
    published_at TIMESTAMPTZ,
    deadline TIMESTAMPTZ,

    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    UNIQUE(class_id, unit_id, assessment_number, academic_period_id, version)
);

-- ============================================================
-- 11. ASSESSMENT TASKS / ITEMS
-- ============================================================
CREATE TABLE assessment_tasks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    task_number INTEGER NOT NULL,
    task_title VARCHAR(255) NOT NULL,
    task_description TEXT,
    max_marks INTEGER NOT NULL DEFAULT 0,
    task_type VARCHAR(30) DEFAULT 'practical' CHECK (task_type IN (
        'practical', 'oral', 'theory', 'observation', 'continuous_test'
    )),
    display_order INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================
-- 12. OBSERVATION CHECKLIST CRITERIA
-- ============================================================
CREATE TABLE checklist_criteria (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    task_id UUID REFERENCES assessment_tasks(id) ON DELETE CASCADE,

    criterion_number VARCHAR(20), -- e.g., "1", "2", "3"
    criterion_text TEXT NOT NULL, -- e.g., "Fixed components firmly and level"
    sub_criterion_text TEXT, -- e.g., "(Any 3 x 1)"

    max_marks INTEGER NOT NULL DEFAULT 0,
    scoring_type VARCHAR(20) DEFAULT 'numerical' CHECK (scoring_type IN (
        'binary', 'numerical', 'award_max'
    )),
    award_max INTEGER, -- For "Award maximum X marks" items
    binary_pass_mark INTEGER DEFAULT 1, -- 1 = pass, 0 = fail

    display_order INTEGER DEFAULT 0,
    is_sub_total BOOLEAN DEFAULT FALSE,
    is_grand_total BOOLEAN DEFAULT FALSE,

    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================
-- 13. ASSESSMENT SESSIONS (Individual Trainee Attempts)
-- ============================================================
CREATE TABLE assessment_sessions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    trainee_id UUID REFERENCES trainees(id) ON DELETE CASCADE,
    assessor_id UUID REFERENCES users(id),

    session_number INTEGER DEFAULT 1, -- For reassessments
    assessment_date DATE,
    assessment_venue VARCHAR(255),
    academic_period_id UUID REFERENCES academic_periods(id),

    total_marks_available INTEGER DEFAULT 0,
    total_marks_obtained DECIMAL(10,2) DEFAULT 0,
    percentage_score DECIMAL(5,2) DEFAULT 0,

    competency_outcome VARCHAR(30) CHECK (competency_outcome IN (
        'competent', 'not_yet_competent', 'proficient', 'mastery', 'pending'
    )) DEFAULT 'pending',

    assessor_comments TEXT,
    feedback_to_candidate TEXT,

    status VARCHAR(20) DEFAULT 'in_progress' CHECK (status IN (
        'in_progress', 'submitted', 'verified', 'returned', 'rejected', 'approved'
    )),

    submitted_at TIMESTAMPTZ,
    verified_at TIMESTAMPTZ,
    approved_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    UNIQUE(assessment_id, trainee_id, session_number)
);

-- ============================================================
-- 14. CRITERION RESULTS (Marks per criterion)
-- ============================================================
CREATE TABLE criterion_results (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    session_id UUID REFERENCES assessment_sessions(id) ON DELETE CASCADE,
    criterion_id UUID REFERENCES checklist_criteria(id) ON DELETE CASCADE,

    marks_available INTEGER NOT NULL,
    marks_obtained DECIMAL(10,2) DEFAULT 0,
    is_competent BOOLEAN,

    assessor_comment TEXT,
    evidence_attached BOOLEAN DEFAULT FALSE,

    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),

    UNIQUE(session_id, criterion_id)
);

-- ============================================================
-- 15. MULTIMEDIA EVIDENCE
-- ============================================================
CREATE TABLE evidence_files (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    session_id UUID REFERENCES assessment_sessions(id) ON DELETE CASCADE,
    criterion_id UUID REFERENCES checklist_criteria(id),
    uploaded_by UUID REFERENCES users(id),

    file_type VARCHAR(20) NOT NULL CHECK (file_type IN (
        'photo', 'video', 'audio', 'document', 'scanned'

    mime_type VARCHAR(100),
    file_size_bytes BIGINT,

    -- Auto-generated filename components stored for reference
    admission_number VARCHAR(50),
    unit_code VARCHAR(50),
    document_type VARCHAR(50),
    assessment_number VARCHAR(50),
    term INTEGER,
    year INTEGER,
    assessed_by_name VARCHAR(200),
    item_reference VARCHAR(50),

    version INTEGER DEFAULT 1,
    is_current BOOLEAN DEFAULT TRUE,

    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ============================================================

CREATE TABLE verification_records (

-- RLS POLICIES (Examples)
-- ============================================================

-- Institutions: all authenticated users can read
CREATE POLICY "institutions_select" ON institutions
    FOR SELECT TO authenticated USING (is_active = TRUE);

-- Users can read their own profile
CREATE POLICY "users_select_own" ON users
    FOR SELECT TO authenticated USING (id = auth.uid());

-- Trainees: trainers and assessors can read trainees in their classes
CREATE POLICY "trainees_select" ON trainees
    FOR SELECT TO authenticated USING (
        EXISTS (
            SELECT 1 FROM trainer_allocations ta
            WHERE ta.class_id = trainees.class_id
            AND ta.trainer_id = auth.uid()
        )
        OR EXISTS (
            SELECT 1 FROM users u
            WHERE u.id = auth.uid()
            AND u.role IN ('super_admin', 'hod', 'management')
        )
    );

-- Assessment sessions: assessors can create/update their own sessions
CREATE POLICY "sessions_insert" ON assessment_sessions
    FOR INSERT TO authenticated WITH CHECK (assessor_id = auth.uid());

CREATE POLICY "sessions_update_own" ON assessment_sessions
    FOR UPDATE TO authenticated USING (assessor_id = auth.uid());

-- Evidence files: only authenticated users can upload
CREATE POLICY "evidence_insert" ON evidence_files
    FOR INSERT TO authenticated WITH CHECK (uploaded_by = auth.uid());

-- Audit logs: read-only for admins
CREATE POLICY "audit_select_admin" ON audit_logs
    FOR SELECT TO authenticated USING (
        EXISTS (
            SELECT 1 FROM users u
            WHERE u.id = auth.uid()
            AND u.role IN ('super_admin', 'management')
        )
    );

-- ============================================================
-- TRIGGERS: Auto-update timestamps
-- ============================================================
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_institutions_updated
    BEFORE UPDATE ON institutions FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_departments_updated
    BEFORE UPDATE ON departments FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_users_updated
    BEFORE UPDATE ON users FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_trainees_updated
    BEFORE UPDATE ON trainees FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_assessments_updated
    BEFORE UPDATE ON assessments FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_sessions_updated
    BEFORE UPDATE ON assessment_sessions FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_criterion_results_updated
    BEFORE UPDATE ON criterion_results FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

Main FastAPI Application

    auth, institutions, departments, courses, classes,

security = HTTPBearer()


async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db)
) -> User:
    """Extract and validate the current user from JWT token."""
    token = credentials.credentials

    try:
        # Verify JWT with Supabase
        user_data = supabase.auth.get_user(token)
        if not user_data or not user_data.user:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid authentication token"
            )

        # Get user from database
        user = db.query(User).filter(User.id == user_data.user.id).first()
        if not user:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="User not found in database"
            )

        if not user.is_active:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="User account is deactivated"
            )

        return user

    except Exception as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Authentication failed: {str(e)}"
        )


def require_role(*roles: str):
    """Dependency factory to require specific user roles."""
    def role_checker(current_user: User = Depends(get_current_user)) -> User:
        if current_user.role not in roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Access denied. Required role(s): {', '.join(roles)}"
            )
        return current_user
    return role_checker

    @staticmethod
    def publish_assessment(
        db: Session,
        assessment_id: uuid.UUID,
        user_id: uuid.UUID
    ) -> Assessment:
        """Publish an assessment — makes it available to assessors."""
        assessment = db.query(Assessment).filter(Assessment.id == assessment_id).first()
        if not assessment:
            raise ValueError("Assessment not found")

        if assessment.status != 'draft':
            raise ValueError("Only draft assessments can be published")

        # Validate: must have at least one task and one criterion
        task_count = db.query(AssessmentTask).filter(
            AssessmentTask.assessment_id == assessment_id
        ).count()
        criterion_count = db.query(ChecklistCriterion).filter(
            ChecklistCriterion.assessment_id == assessment_id
        ).count()

        if task_count == 0:
            raise ValueError("Assessment must have at least one task before publishing")
        if criterion_count == 0:
            raise ValueError("Assessment must have at least one checklist criterion")

        # Calculate total marks
        total_marks = db.query(
            db.func.coalesce(db.func.sum(ChecklistCriterion.max_marks), 0)
        ).filter(ChecklistCriterion.assessment_id == assessment_id).scalar()

        assessment.status = 'published'
        assessment.published_at = datetime.utcnow()
        assessment.total_marks = total_marks
        db.commit()
        db.refresh(assessment)

        # Audit
        audit = AuditLog(
            user_id=user_id,
            action="assessment_published",
            entity_type="assessment",
            entity_id=assessment_id,
            new_values={"total_marks": total_marks}
        )
        db.add(audit)
        db.commit()

        return assessment

                raise ValueError(

                )
            return existing

        # Get assessment to determine total marks available
        assessment = db.query(Assessment).filter(Assessment.id == assessment_id).first()
        if not assessment:
            raise ValueError("Assessment not found")

        total_available = db.query(

        ).filter(ChecklistCriterion.assessment_id == assessment_id).scalar()

        session = AssessmentSession(

            assessor_id=assessor_id,
            session_number=session_number,
            assessment_date=assessment_date,

            academic_period_id=assessment.academic_period_id,
            total_marks_available=total_available,
            status='in_progress'
        )
        db.add(session)
        db.commit()
        db.refresh(session)
        return session

    @staticmethod
    def record_criterion_mark(
        db: Session,
        session_id: uuid.UUID,
        criterion_id: uuid.UUID,
        marks_obtained: float,
        assessor_comment: Optional[str] = None,
        is_competent: Optional[bool] = None,
        user_id: uuid.UUID = None
    ) -> CriterionResult:
        """Record marks for a single criterion."""
        criterion = db.query(ChecklistCriterion).filter(
            ChecklistCriterion.id == criterion_id
        ).first()
        if not criterion:
            raise ValueError("Checklist criterion not found")

        # Validate marks
        if marks_obtained < 0:
            raise ValueError("Marks cannot be negative")
        if marks_obtained > criterion.max_marks:
            raise ValueError(
                f"Marks obtained ({marks_obtained}) cannot exceed "
                f"maximum ({criterion.max_marks})"
            )

        # Determine competency for binary scoring
        if criterion.scoring_type == 'binary':
            is_competent = marks_obtained >= criterion.binary_pass_mark

        # Upsert criterion result
        result = db.query(CriterionResult).filter(
            CriterionResult.session_id == session_id,
            CriterionResult.criterion_id == criterion_id
        ).first()

        if result:
            old_marks = result.marks_obtained
            result.marks_obtained = marks_obtained
            result.assessor_comment = assessor_comment
            result.is_competent = is_competent
        else:
            old_marks = 0
            result = CriterionResult(
                session_id=session_id,
                criterion_id=criterion_id,
                marks_available=criterion.max_marks,
                marks_obtained=marks_obtained,
                assessor_comment=assessor_comment,
                is_competent=is_competent
            )
            db.add(result)

        db.commit()
        db.refresh(result)

        # Audit
        if user_id:
            audit = AuditLog(
                user_id=user_id,
                action="criterion_mark_recorded",
                entity_type="criterion_result",
                entity_id=result.id,
                old_values={"marks_obtained": old_marks},
                new_values={"marks_obtained": marks_obtained}
            )
            db.add(audit)
            db.commit()

        return result

    @staticmethod
    def calculate_session_totals(db: Session, session_id: uuid.UUID) -> Dict:
        """Calculate total marks, percentage, and competency for a session."""
        session = db.query(AssessmentSession).filter(
            AssessmentSession.id == session_id
        ).first()
        if not session:
            raise ValueError("Session not found")

        # Sum all criterion marks
        total_obtained = db.query(
            func.coalesce(func.sum(CriterionResult.marks_obtained), 0)
        ).filter(CriterionResult.session_id == session_id).scalar()

        total_available = session.total_marks_available
        percentage = (total_obtained / total_available * 100) if total_available > 0 else 0

        # Get assessment passing percentage
        assessment = db.query(Assessment).filter(Assessment.id == session.assessment_id).first()
        passing_pct = assessment.passing_percentage if assessment else 50.0

        # Determine competency outcome
        if percentage >= passing_pct:
            competency = 'competent'
        else:
            competency = 'not_yet_competent'

        # Check for mastery (e.g., >= 80%)
        if percentage >= 80:
            competency = 'mastery'
        elif percentage >= 70:
            competency = 'proficient'

        session.total_marks_obtained = round(total_obtained, 2)
        session.percentage_score = round(percentage, 2)
        session.competency_outcome = competency

        db.commit()
        db.refresh(session)

        return {
            "session_id": str(session.id),
            "total_obtained": total_obtained,
            "total_available": total_available,
            "percentage": round(percentage, 2),
            "competency": competency,
            "passing_percentage": passing_pct
        }

class MarksheetService:

    def __init__(self, db: Session):
        self.db = db

    def generate_marksheet(
        self,
        class_id: uuid.UUID,
        unit_id: uuid.UUID,
        academic_period_id: uuid.UUID,
        output_format: str = 'excel'  # 'excel', 'pdf', 'csv'
    ) -> Dict:
        """
        Generate consolidated marksheet with oral and practical averages.
        """
        # Get class, unit info
        class_obj = self.db.query(Class).filter(Class.id == class_id).first()
        unit = self.db.query(Unit).filter(Unit.id == unit_id).first()

        # Get all trainees in the class
        trainees = self.db.query(Trainee).filter(
            Trainee.class_id == class_id,
            Trainee.status == 'active'
        ).order_by(Trainee.admission_number).all()

        # Get all assessments for this class/unit/period
        assessments = self.db.query(Assessment).filter(
            Assessment.class_id == class_id,
            Assessment.unit_id == unit_id,
            Assessment.academic_period_id == academic_period_id,
            Assessment.status.in_(['published', 'archived'])
        ).order_by(Assessment.assessment_type, Assessment.assessment_number).all()

        # Separate oral and practical
        oral_assessments = [a for a in assessments if a.assessment_type == 'oral']
        practical_assessments = [a for a in assessments if a.assessment_type == 'practical']

        # Build marksheet data
        marksheet_data = []

        for trainee in trainees:
            row = {
                'admission_number': trainee.admission_number,
                'trainee_name': f"{trainee.first_name} {trainee.last_name}",
                'oral_marks': {},
                'practical_marks': {},
            }

            # Get oral assessment results
            oral_total = 0
            oral_count = 0
            for oa in oral_assessments:
                session = self.db.query(AssessmentSession).filter(
                    AssessmentSession.assessment_id == oa.id,
                    AssessmentSession.trainee_id == trainee.id,
                    AssessmentSession.status.in_(['submitted', 'verified', 'approved'])
                ).first()

                if session:
                    row['oral_marks'][oa.assessment_number] = {
                        'marks_obtained': session.total_marks_obtained,
                        'marks_available': session.total_marks_available,
                        'percentage': session.percentage_score,
                        'competency': session.competency_outcome
                    }
                    oral_total += session.percentage_score
                    oral_count += 1
                else:
                    row['oral_marks'][oa.assessment_number] = None

            row['oral_average'] = (oral_total / oral_count) if oral_count > 0 else None
            row['oral_count'] = oral_count

            # Get practical assessment results
            prac_total = 0
            prac_count = 0
            for pa in practical_assessments:
                session = self.db.query(AssessmentSession).filter(
                    AssessmentSession.assessment_id == pa.id,
                    AssessmentSession.trainee_id == trainee.id,
                    AssessmentSession.status.in_(['submitted', 'verified', 'approved'])
                ).first()

                if session:
                    row['practical_marks'][pa.assessment_number] = {
                        'marks_obtained': session.total_marks_obtained,
                        'marks_available': session.total_marks_available,
                        'percentage': session.percentage_score,
                        'competency': session.competency_outcome
                    }
                    prac_total += session.percentage_score
                    prac_count += 1
                else:
                    row['practical_marks'][pa.assessment_number] = None

            row['practical_average'] = (prac_total / prac_count) if prac_count > 0 else None
            row['practical_count'] = prac_count

            # Overall average (if both exist)
            if row['oral_average'] is not None and row['practical_average'] is not None:
                row['overall_average'] = (row['oral_average'] + row['practical_average']) / 2
            elif row['oral_average'] is not None:
                row['overall_average'] = row['oral_average']
            elif row['practical_average'] is not None:
                row['overall_average'] = row['practical_average']
            else:
                row['overall_average'] = None

            marksheet_data.append(row)

        # Generate output
        if output_format == 'excel':
            return self._generate_excel_marksheet(
                marksheet_data, oral_assessments, practical_assessments,
                class_obj, unit, academic_period_id
            )
        elif output_format == 'csv':
            return self._generate_csv_marksheet(
                marksheet_data, oral_assessments, practical_assessments,
                class_obj, unit
            )
        else:
            raise ValueError(f"Unsupported format: {output_format}")

                competency = "Competent" if overall >= 50 else "Not Yet Competent"
            else:
                competency = "Pending"

            "filename": filename,

"""
Standardized File Naming Convention Service.
Format: ADM_NO-UNIT_NAME-DOCUMENT_TYPE-ASSESSMENT_NUMBER-TERM-YEAR-ASSESSED_BY-TRAINER_NAME
"""

import re
from typing import Optional


class FilenameService:

    @staticmethod
    def _sanitize(text: str) -> str:
        """Sanitize text for use in filenames."""
        # Remove special characters, replace spaces with underscores
        text = re.sub(r'[^\w\s-]', '', text)
        text = re.sub(r'[\s]+', '_', text.strip())
        return text.upper()

    @classmethod
    def generate_assessment_filename(
        cls,
        admission_number: str,
        unit_name: str,
        document_type: str,
        assessment_number: str,
        term: int,
        year: int,
        assessed_by_name: str,
        item_reference: Optional[str] = None,
        file_index: Optional[int] = None,
        extension: str = "pdf"
    ) -> str:
        """
        Generate standardized filename for assessment documents.
        Example: EEN_2027_001-CONTROL_SYSTEMS_ENGINEERING-OBSERVATION_CHECKLIST-AS01-TERM1-2027-ASSESSED_BY-LAWRENCE_KASITET.pdf
        """
        parts = [
            cls._sanitize(admission_number),
            cls._sanitize(unit_name),
            cls._sanitize(document_type),
            cls._sanitize(assessment_number),
            f"TERM{term}",
            str(year),
            "ASSESSED_BY",
            cls._sanitize(assessed_by_name)
        ]

        if item_reference:
            parts.append(cls._sanitize(item_reference))

        if file_index is not None:
            parts.append(f"FILE{file_index:02d}")

        return "-".join(parts) + f".{extension}"

    @classmethod
    def generate_evidence_filename(
        cls,
        admission_number: str,
        unit_code: str,
        document_type: str,  # PHOTO, VIDEO, AUDIO, DOCUMENT, SCANNED
        assessment_number: str,
        term: int,
        year: int,
        assessed_by_name: str,
        item_reference: Optional[str] = None,
        file_index: int = 1,
        extension: str = "jpg"
    ) -> str:
        """
        Generate standardized filename for evidence files.
        Example: EEN_2027_001-CONTROL_SYSTEMS_ENGINEERING-PHOTO-AS01-TERM1-2027-ASSESSED_BY-LAWRENCE_KASITET-IT01-FILE01.jpg
        """
        return cls.generate_assessment_filename(
            admission_number=admission_number,
            unit_name=unit_code,
            document_type=document_type,
            assessment_number=assessment_number,
            term=term,
            year=year,
            assessed_by_name=assessed_by_name,
            item_reference=item_reference,
            file_index=file_index,
            extension=extension
        )

    @classmethod
    def generate_marksheet_filename(
        cls,
        unit_code: str,
        class_name: str,
        academic_period: str,
        extension: str = "xlsx"
    ) -> str:
        """Generate filename for marksheets."""
        return f"MARKSHEET-{cls._sanitize(unit_code)}-{cls._sanitize(class_name)}-{cls._sanitize(academic_period)}.{extension}"

    @classmethod
    def generate_report_filename(
        cls,
        report_type: str,
        department: str,
        academic_period: str,
        extension: str = "pdf"
    ) -> str:
        """Generate filename for reports."""
        return f"REPORT-{cls._sanitize(report_type)}-{cls._sanitize(department)}-{cls._sanitize(academic_period)}.{extension}"

class EvidenceService:

    ALLOWED_TYPES = {
        'photo': settings.ALLOWED_IMAGE_TYPES,
        'video': settings.ALLOWED_VIDEO_TYPES,
        'audio': settings.ALLOWED_AUDIO_TYPES,
        'document': settings.ALLOWED_DOC_TYPES,
        'scanned': settings.ALLOWED_IMAGE_TYPES + ['application/pdf']
    }

    MAX_FILE_SIZE = settings.MAX_FILE_SIZE_MB * 1024 * 1024  # Convert to bytes

    @staticmethod
    def _detect_file_type(mime_type: str) -> str:
        """Detect evidence category from MIME type."""
        for category, types in EvidenceService.ALLOWED_TYPES.items():
            if mime_type in types:
                return category
        return 'document'

    @classmethod
    def upload_evidence(
        cls,
        db: Session,
        session_id: uuid.UUID,
        criterion_id: Optional[uuid.UUID],
        file_content: bytes,
        original_filename: str,
        uploaded_by: uuid.UUID,
        item_reference: Optional[str] = None
    ) -> EvidenceFile:
        """Upload multimedia evidence for an assessment criterion."""
        # Validate session
        session = db.query(AssessmentSession).filter(
            AssessmentSession.id == session_id
        ).first()
        if not session:
            raise ValueError("Assessment session not found")

        # Get trainee and assessment info for filename
        trainee = db.query(Trainee).filter(Trainee.id == session.trainee_id).first()
        assessment = db.query(Assessment).filter(Assessment.id == session.assessment_id).first()
        assessor = db.query(User).filter(User.id == uploaded_by).first()

        # Detect file type
        mime_type, _ = mimetypes.guess_type(original_filename)
        if not mime_type:
            mime_type = 'application/octet-stream'

        file_type = cls._detect_file_type(mime_type)

        # Validate file size
        if len(file_content) > cls.MAX_FILE_SIZE:
            raise ValueError(f"File exceeds maximum size of {settings.MAX_FILE_SIZE_MB}MB")

        # Validate MIME type
        if mime_type not in cls.ALLOWED_TYPES.get(file_type, []):
            raise ValueError(f"File type {mime_type} not allowed for category {file_type}")

        # Get extension
        ext = original_filename.rsplit('.', 1)[-1] if '.' in original_filename else 'bin'

        # Generate system filename
        system_filename = FilenameService.generate_evidence_filename(
            admission_number=trainee.admission_number,
            unit_code=assessment.unit.code if assessment.unit else "UNKNOWN",
            document_type=file_type.upper(),

            assessed_by_name=f"{assessor.first_name}_{assessor.last_name}" if assessor else "Unknown",
            item_reference=item_reference,
            file_index=1,
            extension=ext
        )

        # Check for existing files with same reference (versioning)
        existing_files = db.query(EvidenceFile).filter(
            EvidenceFile.session_id == session_id,
            EvidenceFile.criterion_id == criterion_id,
            EvidenceFile.is_current == True

        file_index = len(existing_files) + 1

        system_filename = FilenameService.generate_evidence_filename(
            admission_number=trainee.admission_number,
            unit_code=assessment.unit.code if assessment.unit else "UNKNOWN",
            document_type=file_type.upper(),

            item_reference=item_reference,
            file_index=file_index,
            extension=ext
        )

        storage_path = f"evidence/{session_id}/{system_filename}"

        try:
            supabase.storage.from_(settings.SUPABASE_STORAGE_BUCKET).upload(
                path=storage_path,
                file=file_content,
                file_options={"content-type": mime_type}
            )
        except Exception as e:
            raise ValueError(f"Failed to upload file to storage: {str(e)}")

        # Mark previous versions as not current
        for ef in existing_files:
            ef.is_current = False
            ef.version += 1

        # Create evidence record
        evidence = EvidenceFile(
            session_id=session_id,
            criterion_id=criterion_id,
            uploaded_by=uploaded_by,
            original_filename=original_filename,

            assessed_by_name=f"{assessor.first_name} {assessor.last_name}" if assessor else None,
            item_reference=item_reference,
            version=1,

"""Internal and External Verification Service."""

from sqlalchemy.orm import Session
from datetime import datetime
from typing import Optional
import uuid

from app.models.session import AssessmentSession
from app.models.verification import VerificationRecord
from app.models.user import User
from app.models.audit import AuditLog


class VerificationService:

    @staticmethod
    def verify_assessment(
        db: Session,
        session_id: uuid.UUID,
        verifier_id: uuid.UUID,
        verifier_type: str,  # 'internal' or 'external'
        decision: str,  # 'approved', 'returned_for_correction', 'rejected'
        comments: Optional[str] = None,
        rejection_reasons: Optional[str] = None
    ) -> VerificationRecord:
        """
        Verify an assessment session.
        Prevents assessors from verifying their own assessments.
        """
        session = db.query(AssessmentSession).filter(
            AssessmentSession.id == session_id
        ).first()
        if not session:
            raise ValueError("Assessment session not found")

        # Prevent self-verification
        if session.assessor_id == verifier_id:
            raise ValueError("Assessors cannot verify their own assessments")

        # Check verifier role
        verifier = db.query(User).filter(User.id == verifier_id).first()
        if verifier_type == 'internal' and verifier.role not in ['super_admin', 'internal_verifier', 'hod']:
            raise ValueError("User is not authorized as internal verifier")
        if verifier_type == 'external' and verifier.role != 'external_verifier':
            raise ValueError("User is not authorized as external verifier")

        # Check session status
        if session.status not in ['submitted', 'returned']:
            raise ValueError(f"Cannot verify session with status: {session.status}")

        # Create verification record
        record = VerificationRecord(
            session_id=session_id,
            verifier_id=verifier_id,
            verifier_type=verifier_type,
            decision=decision,
            comments=comments,
            rejection_reasons=rejection_reasons,
            verified_at=datetime.utcnow()
        )
        db.add(record)

        # Update session status
        if decision == 'approved':
            session.status = 'approved'
            session.approved_at = datetime.utcnow()
        elif decision == 'returned_for_correction':
            session.status = 'returned'
        elif decision == 'rejected':
            session.status = 'rejected'

        db.commit()
        db.refresh(record)

        # Audit log
        audit = AuditLog(
            user_id=verifier_id,
            action=f"assessment_{decision}",
            entity_type="verification",
            entity_id=record.id,
            new_values={
                "session_id": str(session_id),
                "verifier_type": verifier_type,
                "decision": decision,
                "comments": comments
            }
        )
        db.add(audit)
        db.commit()

        return record

    @staticmethod
    def get_verification_history(
        db: Session,
        session_id: uuid.UUID
    ) -> list:
        """Get complete verification history for a session."""
        records = db.query(VerificationRecord).filter(
            VerificationRecord.session_id == session_id
        ).order_by(VerificationRecord.verified_at.desc()).all()

        result = []
        for record in records:
            verifier = db.query(User).filter(User.id == record.verifier_id).first()
            result.append({
                "verification_id": str(record.id),
                "verifier_name": f"{verifier.first_name} {verifier.last_name}" if verifier else "Unknown",
                "verifier_type": record.verifier_type,
                "decision": record.decision,
                "comments": record.comments,
                "rejection_reasons": record.rejection_reasons,
                "verified_at": record.verified_at.isoformat() if record.verified_at else None
            })

        return result

    try {
      const marksPayload = Object.entries(results).map(([criterionId, result]) => ({
        criterion_id: criterionId,
        marks_obtained: result.marks_obtained,
        assessor_comment: comments[criterionId] || undefined,
      }));

      await apiClient.post(`/api/v1/marking/sessions/${sessionId}/marks`, marksPayload);
      setSuccess('Marks saved successfully');
      setTimeout(() => setSuccess(null), 3000);
    } catch (err: any) {
      setError(err.response?.data?.detail || 'Failed to save marks');
    } finally {
      setSaving(false);
    }
  };

  const submitAssessment = async () => {
    if (!window.confirm('Submit this assessment for verification? This action cannot be undone.')) {
      return;
    }

    setSubmitting(true);
    setError(null);
    try {
      await apiClient.post(`/api/v1/marking/sessions/${sessionId}/submit`, {
        assessor_comments: assessorComments,
        feedback_to_candidate: feedbackToCandidate,
      });
      setSuccess('Assessment submitted for verification');
      setTimeout(() => navigate('/assessments'), 2000);
    } catch (err: any) {
      setError(err.response?.data?.detail || 'Failed to submit assessment');
    } finally {
      setSubmitting(false);
    }
  };

  const calculateTotals = () => {
    const totalObtained = Object.values(results).reduce((sum, r) => sum + (r.marks_obtained || 0), 0);
    const totalAvailable = criteria.reduce((sum, c) => sum + (c.is_sub_total || c.is_grand_total ? 0 : c.max_marks), 0);
    const percentage = totalAvailable > 0 ? (totalObtained / totalAvailable) * 100 : 0;
    return { totalObtained, totalAvailable, percentage };
  };

  const { totalObtained, totalAvailable, percentage } = calculateTotals();

  const getCompetencyColor = (outcome: string) => {
    switch (outcome) {
      case 'competent': return 'bg-green-100 text-green-800';
      case 'not_yet_competent': return 'bg-red-100 text-red-800';
      case 'proficient': return 'bg-blue-100 text-blue-800';
      case 'mastery': return 'bg-purple-100 text-purple-800';
      default: return 'bg-gray-100 text-gray-800';
    }
  };

  if (error) {
    return (
      <Alert variant="destructive">
        <AlertCircle className="h-4 w-4" />
        <AlertDescription>{error}</AlertDescription>
      </Alert>
    );
  }

  if (!session || !assessment) {
    return <div className="p-8 text-center">Loading assessment...</div>;
  }

  return (
    <div className="space-y-6 p-6">
      {/* Header */}
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold">{assessment.title}</h1>
          <p className="text-muted-foreground">
            Assessment #{assessment.assessment_number} | {assessment.assessment_type.toUpperCase()}
          </p>
        </div>
        <Badge className={getCompetencyColor(session.competency_outcome)}>
          {session.competency_outcome.replace('_', ' ').toUpperCase()}
        </Badge>
      </div>

      {/* Progress Summary */}
      <Card>
        <CardContent className="pt-6">
          <div className="grid grid-cols-4 gap-4">
            <div className="text-center">
              <p className="text-sm text-muted-foreground">Marks Obtained</p>
              <p className="text-2xl font-bold">{totalObtained.toFixed(1)}</p>
            </div>
            <div className="text-center">
              <p className="text-sm text-muted-foreground">Total Available</p>
              <p className="text-2xl font-bold">{totalAvailable}</p>
            </div>
            <div className="text-center">
              <p className="text-sm text-muted-foreground">Percentage</p>
              <p className="text-2xl font-bold">{percentage.toFixed(1)}%</p>
            </div>
            <div className="text-center">
              <p className="text-sm text-muted-foreground">Passing Mark</p>
              <p className="text-2xl font-bold">{assessment.passing_percentage}%</p>
            </div>
          </div>
          <Progress value={percentage} className="mt-4" />
        </CardContent>
      </Card>

      {/* Success/Error Messages */}
      {success && (
        <Alert className="bg-green-50 border-green-200">
          <CheckCircle className="h-4 w-4 text-green-600" />
          <AlertDescription className="text-green-800">{success}</AlertDescription>
        </Alert>
      )}

      {/* Tabs: Checklist | Evidence | Comments */}
      <Tabs defaultValue="checklist">
        <TabsList>
          <TabsTrigger value="checklist">Observation Checklist</TabsTrigger>
          <TabsTrigger value="evidence">Evidence</TabsTrigger>
          <TabsTrigger value="comments">Comments & Submit</TabsTrigger>
        </TabsList>

        {/* OBSERVATION CHECKLIST TAB */}
        <TabsContent value="checklist">
          <Card>
            <CardHeader>
              <CardTitle>Observation Checklist</CardTitle>
            </CardHeader>
            <CardContent>
              <Table>
                <TableHeader>
                  <TableRow>
                    <TableHead className="w-[40%]">Item to be Evaluated</TableHead>
                    <TableHead className="w-[15%] text-center">Marks Available</TableHead>
                    <TableHead className="w-[20%] text-center">Marks Obtained</TableHead>
                    <TableHead className="w-[25%]">Comments</TableHead>
                  </TableRow>
                </TableHeader>
                <TableBody>
                  {criteria.map((criterion) => {
                    if (criterion.is_sub_total) {
                      const subTotal = Object.entries(results)
                        .filter(([cid]) => {
                          const c = criteria.find(x => x.id === cid);
                          return c && !c.is_sub_total && !c.is_grand_total;
                        })
                        .reduce((sum, [, r]) => sum + r.marks_obtained, 0);

                      return (
                        <TableRow key={criterion.id} className="bg-muted font-bold">
                          <TableCell>Sub-Total {criterion.criterion_number}</TableCell>
                          <TableCell className="text-center">{criterion.max_marks}</TableCell>
                          <TableCell className="text-center">{subTotal.toFixed(1)}</TableCell>
                          <TableCell></TableCell>
                        </TableRow>
                      );
                    }

                    if (criterion.is_grand_total) {
                      return (
                        <TableRow key={criterion.id} className="bg-primary text-primary-foreground font-bold">
                          <TableCell>GRAND TOTAL</TableCell>
                          <TableCell className="text-center">{criterion.max_marks}</TableCell>
                          <TableCell className="text-center">{totalObtained.toFixed(1)}</TableCell>
                          <TableCell></TableCell>
                        </TableRow>
                      );
                    }

                    const result = results[criterion.id];
                    const currentMark = result?.marks_obtained || 0;

                    return (
                      <TableRow key={criterion.id}>
                        <TableCell>
                          <div>
                            <p className="font-medium">{criterion.criterion_text}</p>
                            {criterion.sub_criterion_text && (
                              <p className="text-sm text-muted-foreground italic">
                                ({criterion.sub_criterion_text})
                              </p>
                            )}
                            {criterion.scoring_type === 'binary' && (
                              <Badge variant="outline" className="mt-1">Binary (1/0)</Badge>
                            )}
                          </div>
                        </TableCell>
                        <TableCell className="text-center font-medium">
                          {criterion.max_marks}
                        </TableCell>
                        <TableCell className="text-center">
                          <Input
                            type="number"
                            min={0}
                            max={criterion.max_marks}
                            step={criterion.scoring_type === 'binary' ? 1 : 0.5}
                            value={currentMark}
                            onChange={(e) => handleMarkChange(
                              criterion.id,
                              parseFloat(e.target.value) || 0,
                              criterion.max_marks
                            )}
                            className="w-20 mx-auto text-center"
                            disabled={session.status !== 'in_progress'}
                          />
                        </TableCell>
                        <TableCell>
                          <Input
                            placeholder="Add comment..."
                            value={comments[criterion.id] || ''}
                            onChange={(e) => handleCommentChange(criterion.id, e.target.value)}
                            disabled={session.status !== 'in_progress'}
                          />
                        </TableCell>
                      </TableRow>
                    );
                  })}
                </TableBody>
              </Table>
            </CardContent>
          </Card>
        </TabsContent>

        {/* EVIDENCE TAB */}
        <TabsContent value="evidence">
          <Card>
            <CardHeader>
              <CardTitle>Multimedia Evidence</CardTitle>
            </CardHeader>
            <CardContent>
              <EvidenceUploader sessionId={sessionId} criteria={criteria} />
            </CardContent>
          </Card>
        </TabsContent>

        {/* COMMENTS & SUBMIT TAB */}
        <TabsContent value="comments">
          <Card>
            <CardHeader>
              <CardTitle>Assessor Comments & Submission</CardTitle>
            </CardHeader>
            <CardContent className="space-y-4">
              <div>
                <label className="text-sm font-medium">Assessor's Overall Comments</label>
                <Textarea
                  value={assessorComments}
                  onChange={(e) => setAssessorComments(e.target.value)}
                  placeholder="Enter overall assessment comments..."
                  rows={4}
                  disabled={session.status !== 'in_progress'}
                />
              </div>
              <div>
                <label className="text-sm font-medium">Feedback to the Candidate</label>
                <Textarea
                  value={feedbackToCandidate}
                  onChange={(e) => setFeedbackToCandidate(e.target.value)}
                  placeholder="Enter feedback for the candidate..."
                  rows={4}
                  disabled={session.status !== 'in_progress'}
                />
              </div>

              <div className="flex gap-4 pt-4">

                  onClick={saveMarks}
                  disabled={saving || session.status !== 'in_progress'}
                  variant="outline"
                >
                  <Save className="mr-2 h-4 w-4" />
                  {saving ? 'Saving...' : 'Save Progress'}

      </Tabs>
    </div>
  );
};
3.5 Evidence Uploader Component (src/components/evidence/EvidenceUploader.tsx)
tsx
import React, { useState, useRef } from 'react';
import { Card, CardContent } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Progress } from '@/components/ui/progress';
import { Upload, X, FileImage, FileVideo, FileAudio, FileText, CheckCircle } from 'lucide-react';
import { apiClient } from '@/api/client';
import type { ChecklistCriterion, EvidenceFile } from '@/types/assessment';

interface EvidenceUploaderProps {
  sessionId: string;
  criteria: ChecklistCriterion[];
}

export const EvidenceUploader: React.FC<EvidenceUploaderProps> = ({ sessionId, criteria }) => {
  const [selectedCriterion, setSelectedCriterion] = useState<string>('');
  const [uploading, setUploading] = useState(false);
  const [uploadProgress, setUploadProgress] = useState(0);
  const [evidence, setEvidence] = useState<EvidenceFile[]>([]);
  const [error, setError] = useState<string | null>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const getFileIcon = (type: string) => {
    switch (type) {
      case 'photo': return <FileImage className="h-5 w-5 text-blue-500" />;
      case 'video': return <FileVideo className="h-5 w-5 text-purple-500" />;
      case 'audio': return <FileAudio className="h-5 w-5 text-orange-500" />;
      default: return <FileText className="h-5 w-5 text-gray-500" />;
    }
  };

  const handleFileSelect = async (event: React.ChangeEvent<HTMLInputElement>) => {
    const files = event.target.files;
    if (!files || files.length === 0) return;

    if (!selectedCriterion) {
      setError('Please select a checklist criterion first');
      return;
    }

    setUploading(true);
    setUploadProgress(0);
    setError(null);

    for (const file of Array.from(files)) {
      try {
        const formData = new FormData();
        formData.append('file', file);
        formData.append('criterion_id', selectedCriterion);

        const response = await apiClient.post(
          `/api/v1/evidence/upload/${sessionId}`,
          formData,
          {
            headers: { 'Content-Type': 'multipart/form-data' },
            onUploadProgress: (progressEvent) => {
              const progress = (progressEvent.loaded / (progressEvent.total || 1)) * 100;
              setUploadProgress(progress);
            }
          }
        );

        setEvidence(prev => [...prev, response.data]);
      } catch (err: any) {
        setError(err.response?.data?.detail || `Failed to upload ${file.name}`);
      }
    }

    setUploading(false);
    setUploadProgress(0);
    if (fileInputRef.current) fileInputRef.current.value = '';
  };

  return (
    <div className="space-y-4">
      {/* Criterion Selector */}
      <div>
        <label className="text-sm font-medium mb-2 block">
          Select Checklist Criterion for Evidence
        </label>
        <select
          value={selectedCriterion}
          onChange={(e) => setSelectedCriterion(e.target.value)}
          className="w-full p-2 border rounded-md"
        >
          <option value="">-- Select Criterion --</option>
          {criteria.filter(c => !c.is_sub_total && !c.is_grand_total).map(c => (
            <option key={c.id} value={c.id}>
              {c.criterion_number}. {c.criterion_text}
            </option>
          ))}
        </select>
      </div>

      {/* Upload Area */}
      <Card className="border-dashed">
        <CardContent className="p-6 text-center">
          <input
            ref={fileInputRef}
            type="file"
            multiple
            accept="image/*,video/*,audio/*,.pdf,.docx,.xlsx"
            onChange={handleFileSelect}
            className="hidden"
          />
          <Upload className="mx-auto h-12 w-12 text-muted-foreground mb-4" />
          <p className="text-sm text-muted-foreground mb-2">
            Drag and drop files here, or click to browse
          </p>
          <p className="text-xs text-muted-foreground">
            Supported: JPG, PNG, WEBP, MP4, MOV, WEBM, MP3, WAV, PDF, DOCX, XLSX
            (Max 100MB per file)
          </p>
          <Button
            onClick={() => fileInputRef.current?.click()}
            disabled={uploading || !selectedCriterion}
            className="mt-4"
          >
            <Upload className="mr-2 h-4 w-4" />
            Select Files
          </Button>
        </CardContent>
      </Card>

      {/* Upload Progress */}
      {uploading && (
        <div>
          <p className="text-sm mb-1">Uploading... {uploadProgress.toFixed(0)}%</p>
          <Progress value={uploadProgress} />
        </div>
      )}

      {/* Error */}
      {error && (
        <p className="text-sm text-red-600">{error}</p>
      )}

      {/* Evidence List */}
      {evidence.length > 0 && (
        <div className="space-y-2">
          <h3 className="font-medium">Uploaded Evidence ({evidence.length})</h3>
          {evidence.map((file) => (
            <div key={file.id} className="flex items-center gap-3 p-3 border rounded-lg">
              {getFileIcon(file.file_type)}
              <div className="flex-1 min-w-0">
                <p className="text-sm font-medium truncate">{file.system_filename}</p>
                <p className="text-xs text-muted-foreground">
                  {file.file_type.toUpperCase()} | {(file.file_size_bytes / 1024).toFixed(1)} KB
                </p>
              </div>
              <Badge variant="secondary">
                v{file.version}
              </Badge>
              <CheckCircle className="h-4 w-4 text-green-500" />
            </div>
          ))}
        </div>
      )}
    </div>
  );
};

# TVET e-ASSESS — Digital Practical Assessment Platform

A centralized web-based platform designed to digitize the entire practical assessment process for TVET institutions.

## Tech Stack
- **Frontend:** React.js + TypeScript + Tailwind CSS + shadcn/ui
- **Backend:** Python FastAPI
- **Database:** Supabase PostgreSQL
- **Storage:** Supabase Storage
- **Hosting:** Render

## Quick Start

### 1. Setup Supabase
1. Create a new Supabase project at https://supabase.com
2. Run `database/schema.sql` in the SQL Editor
3. Copy your project URL and keys

### 2. Backend
```bash
cd backend
python -m venv venv
source venv/bin/activate  # Windows: venv\Scripts\activate
pip install -r requirements.txt
cp .env.example .env  # Edit with your Supabase credentials
uvicorn app.main:app --reload --port 8000
3. Frontend
bash
cd frontend
npm install
cp .env.example .env.local  # Edit with your Supabase keys
npm run dev
4. Access
Frontend: http://localhost:5173
Backend API Docs: http://localhost:8000/api/docs
Project Motto
Assess Digitally. Verify Competence. Preserve Evidence.

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. INSTITUTIONS
CREATE TABLE IF NOT EXISTS institutions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    address TEXT,
    phone VARCHAR(50),
    email VARCHAR(100),
    logo_url TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. DEPARTMENTS
CREATE TABLE IF NOT EXISTS departments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL,
    hod_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(institution_id, code)
);

-- 3. COURSES
CREATE TABLE IF NOT EXISTS courses (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL,
    qualification_level VARCHAR(50),
    duration_months INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(department_id, code)
);

-- 4. CLASSES
CREATE TABLE IF NOT EXISTS classes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    intake_year INTEGER,
    cycle INTEGER,
    level INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, name)
);

-- 5. UNITS
CREATE TABLE IF NOT EXISTS units (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    code VARCHAR(50) NOT NULL,
    name VARCHAR(255) NOT NULL,
    credit_hours INTEGER,
    level INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, code)
);

-- 6. USERS
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    institution_id UUID REFERENCES institutions(id),
    department_id UUID REFERENCES departments(id),
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(50),
    staff_number VARCHAR(50),
    role VARCHAR(50) NOT NULL CHECK (role IN (
        'super_admin', 'hod', 'trainer', 'assessor',
        'internal_verifier', 'external_verifier', 'trainee', 'management'
    )),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. TRAINEES
CREATE TABLE IF NOT EXISTS trainees (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    admission_number VARCHAR(50) UNIQUE NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    national_id VARCHAR(50),
    phone VARCHAR(50),
    email VARCHAR(255),
    enrollment_date DATE,
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'graduated', 'dropped', 'transferred')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 8. TRAINER ALLOCATIONS
CREATE TABLE IF NOT EXISTS trainer_allocations (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    trainer_id UUID REFERENCES users(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    unit_id UUID REFERENCES units(id) ON DELETE CASCADE,
    academic_year INTEGER NOT NULL,
    term INTEGER NOT NULL CHECK (term IN (1, 2, 3)),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(trainer_id, class_id, unit_id, academic_year, term)
);

-- 9. ACADEMIC PERIODS
CREATE TABLE IF NOT EXISTS academic_periods (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    year INTEGER NOT NULL,
    term INTEGER NOT NULL CHECK (term IN (1, 2, 3)),
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(institution_id, year, term)
);

-- 10. ASSESSMENTS
CREATE TABLE IF NOT EXISTS assessments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    unit_id UUID REFERENCES units(id) ON DELETE CASCADE,
    academic_period_id UUID REFERENCES academic_periods(id),
    title VARCHAR(255) NOT NULL,
    assessment_number VARCHAR(50) NOT NULL,
    assessment_type VARCHAR(20) NOT NULL CHECK (assessment_type IN ('oral', 'practical')),
    assessment_level VARCHAR(20),
    instructions TEXT,
    duration_minutes INTEGER,
    total_marks INTEGER,
    passing_percentage DECIMAL(5,2) DEFAULT 50.00,
    required_tools TEXT,
    required_equipment TEXT,
    required_materials TEXT,
    safety_precautions TEXT,
    ppe_requirements TEXT,
    marking_guidelines TEXT,
    reference_documents TEXT,
    status VARCHAR(20) DEFAULT 'draft' CHECK (status IN (
        'draft', 'preview', 'published', 'withdrawn', 'archived'
    )),
    version INTEGER DEFAULT 1,
    published_at TIMESTAMPTZ,
    deadline TIMESTAMPTZ,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(class_id, unit_id, assessment_number, academic_period_id, version)
);

-- 11. ASSESSMENT TASKS
CREATE TABLE IF NOT EXISTS assessment_tasks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    task_number INTEGER NOT NULL,
    task_title VARCHAR(255) NOT NULL,
    task_description TEXT,
    max_marks INTEGER NOT NULL DEFAULT 0,
    task_type VARCHAR(30) DEFAULT 'practical' CHECK (task_type IN (
        'practical', 'oral', 'theory', 'observation', 'continuous_test'
    )),
    display_order INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 12. CHECKLIST CRITERIA
CREATE TABLE IF NOT EXISTS checklist_criteria (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    task_id UUID REFERENCES assessment_tasks(id) ON DELETE CASCADE,
    criterion_number VARCHAR(20),
    criterion_text TEXT NOT NULL,
    sub_criterion_text TEXT,
    max_marks INTEGER NOT NULL DEFAULT 0,
    scoring_type VARCHAR(20) DEFAULT 'numerical' CHECK (scoring_type IN (
        'binary', 'numerical', 'award_max'
    )),
    award_max INTEGER,
    binary_pass_mark INTEGER DEFAULT 1,
    display_order INTEGER DEFAULT 0,
    is_sub_total BOOLEAN DEFAULT FALSE,
    is_grand_total BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 13. ASSESSMENT SESSIONS
CREATE TABLE IF NOT EXISTS assessment_sessions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    trainee_id UUID REFERENCES trainees(id) ON DELETE CASCADE,
    assessor_id UUID REFERENCES users(id),
    session_number INTEGER DEFAULT 1,
    assessment_date DATE,
    assessment_venue VARCHAR(255),
    academic_period_id UUID REFERENCES academic_periods(id),
    total_marks_available INTEGER DEFAULT 0,
    total_marks_obtained DECIMAL(10,2) DEFAULT 0,
    percentage_score DECIMAL(5,2) DEFAULT 0,
    competency_outcome VARCHAR(30) CHECK (competency_outcome IN (
        'competent', 'not_yet_competent', 'proficient', 'mastery', 'pending'
    )) DEFAULT 'pending',
    assessor_comments TEXT,
    feedback_to_candidate TEXT,
    status VARCHAR(20) DEFAULT 'in_progress' CHECK (status IN (
        'in_progress', 'submitted', 'verified', 'returned', 'rejected', 'approved'
    )),
    submitted_at TIMESTAMPTZ,
    verified_at TIMESTAMPTZ,
    approved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(assessment_id, trainee_id, session_number)
);

-- 14. CRITERION RESULTS

);

-- AUTO-UPDATE TRIGGER FUNCTION
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_institutions_updated BEFORE UPDATE ON institutions FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- RLS POLICIES
CREATE POLICY "institutions_select" ON institutions FOR SELECT TO authenticated USING (is_active = TRUE);
CREATE POLICY "users_select_own" ON users FOR SELECT TO authenticated USING (id = auth.uid());

CREATE POLICY "trainees_select" ON trainees FOR SELECT TO authenticated USING (
    EXISTS (SELECT 1 FROM trainer_allocations ta WHERE ta.class_id = trainees.class_id AND ta.trainer_id = auth.uid())
    OR EXISTS (SELECT 1 FROM users u WHERE u.id = auth.uid() AND u.role IN ('super_admin', 'hod', 'management'))
);

CREATE POLICY "sessions_insert" ON assessment_sessions FOR INSERT TO authenticated WITH CHECK (assessor_id = auth.uid());
CREATE POLICY "sessions_update_own" ON assessment_sessions FOR UPDATE TO authenticated USING (assessor_id = auth.uid());
CREATE POLICY "sessions_select" ON assessment_sessions FOR SELECT TO authenticated USING (
    assessor_id = auth.uid()
    OR EXISTS (SELECT 1 FROM users u WHERE u.id = auth.uid() AND u.role IN ('super_admin', 'hod', 'internal_verifier', 'external_verifier', 'management'))
);

CREATE POLICY "evidence_insert" ON evidence_files FOR INSERT TO authenticated WITH CHECK (uploaded_by = auth.uid());
CREATE POLICY "evidence_select" ON evidence_files FOR SELECT TO authenticated USING (TRUE);

CREATE POLICY "audit_select_admin" ON audit_logs FOR SELECT TO authenticated USING (
    EXISTS (SELECT 1 FROM users u WHERE u.id = auth.uid() AND u.role IN ('super_admin', 'management'))

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.orm import Session
from app.database import get_db, supabase
from app.models.user import User

security = HTTPBearer()


async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db),
) -> User:
    token = credentials.credentials
    try:
        user_data = supabase.auth.get_user(token)
        if not user_data or not user_data.user:
            raise HTTPException(status_code=401, detail="Invalid token")
        user = db.query(User).filter(User.id == user_data.user.id).first()
        if not user:
            raise HTTPException(status_code=404, detail="User not found")
        if not user.is_active:
            raise HTTPException(status_code=403, detail="Account deactivated")
        return user
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=401, detail=f"Authentication failed: {str(e)}")


def require_role(*roles: str):
    def role_checker(current_user: User = Depends(get_current_user)) -> User:
        if current_user.role not in roles:
            raise HTTPException(status_code=403, detail=f"Required role(s): {', '.join(roles)}")
        return current_user
    return role_checker

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.orm import Session
from app.database import get_db, supabase
from app.models.user import User
from app.auth.schemas import LoginRequest, LoginResponse, RegisterRequest
from app.auth.dependencies import get_current_user, require_admin

router = APIRouter()


@router.post("/login", response_model=LoginResponse)
def login(data: LoginRequest, db: Session = Depends(get_db)):
    try:
        auth_response = supabase.auth.sign_in_with_password({
            "email": data.email,
            "password": data.password,
        })
        if not auth_response.session:
            raise HTTPException(status_code=401, detail="Invalid credentials")

        user = db.query(User).filter(User.id == auth_response.user.id).first()
        if not user:
            raise HTTPException(status_code=404, detail="User profile not found")

        return LoginResponse(
            access_token=auth_response.session.access_token,
            user_id=str(user.id),
            role=user.role,
            full_name=f"{user.first_name} {user.last_name}",
        )
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=401, detail=str(e))


@router.post("/register")
def register(data: RegisterRequest, db: Session = Depends(get_db), admin: User = Depends(require_admin)):
    try:
        auth_response = supabase.auth.admin.create_user({
            "email": data.email,
            "password": data.password,
            "email_confirm": True,
            "user_metadata": {
                "first_name": data.first_name,
                "last_name": data.last_name,
            },
        })
        user = User(
            id=auth_response.user.id,
            email=data.email,
            first_name=data.first_name,
            last_name=data.last_name,
            role=data.role,
            department_id=data.department_id,
            staff_number=data.staff_number,
            phone=data.phone,
        )
        db.add(user)
        db.commit()
        db.refresh(user)
        return {"message": "User registered", "user_id": str(user.id)}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))


@router.get("/me")
def get_me(current_user: User = Depends(get_current_user)):
    return {
        "id": str(current_user.id),
        "email": current_user.email,
        "first_name": current_user.first_name,
        "last_name": current_user.last_name,
        "role": current_user.role,
        "department_id": str(current_user.department_id) if current_user.department_id else None,
    }


@router.post("/logout")
def logout(current_user: User = Depends(get_current_user)):
    supabase.auth.sign_out()
    return {"message": "Logged out"}

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from typing import List
from pydantic import BaseModel
from app.database import get_db
from app.models.institution import Institution
from app.auth.dependencies import get_current_user, require_admin
from app.models.user import User

router = APIRouter()


class InstitutionCreate(BaseModel):
    name: str
    code: str
    address: str = None
    phone: str = None
    email: str = None


@router.post("/", response_model=dict)
def create_institution(data: InstitutionCreate, db: Session = Depends(get_db), user: User = Depends(require_admin)):
    inst = Institution(**data.model_dump())
    db.add(inst)
    db.commit()
    db.refresh(inst)
    return {"id": str(inst.id), "name": inst.name}


@router.get("/", response_model=List[dict])
def list_institutions(db: Session = Depends(get_db), user: User = Depends(get_current_user)):
    return [{"id": str(i.id), "name": i.name, "code": i.code} for i in db.query(Institution).filter(Institution.is_active == True).all()]

from typing import List
from pydantic import BaseModel
from app.database import get_db
from app.models.trainee import Trainee
from app.auth.dependencies import get_current_user, require_trainer
from app.models.user import User
import uuid

router = APIRouter()


class TraineeCreate(BaseModel):
    class_id: uuid.UUID
    admission_number: str
    first_name: str
    last_name: str
    national_id: str = None
    phone: str = None
    email: str = None


@router.post("/")
def create_trainee(data: TraineeCreate, db: Session = Depends(get_db), user: User = Depends(require_trainer)):
    trainee = Trainee(**data.model_dump())
    db.add(trainee)
    db.commit()
    db.refresh(trainee)
    return {"id": str(trainee.id), "admission_number": trainee.admission_number}


@router.get("/")
def list_trainees(class_id: uuid.UUID = None, db: Session = Depends(get_db), user: User = Depends(require_trainer)):
    q = db.query(Trainee).filter(Trainee.status == 'active')
    if class_id:
        q = q.filter(Trainee.class_id == class_id)
    return [{
        "id": str(t.id),
        "admission_number": t.admission_number,
        "first_name": t.first_name,
        "last_name": t.last_name,
        "class_id": str(t.class_id),
    } for t in q.order_by(Trainee.admission_number).all()]


@router.get("/{trainee_id}")
def get_trainee(trainee_id: uuid.UUID, db: Session = Depends(get_db), user: User = Depends(require_trainer)):
    t = db.query(Trainee).filter(Trainee.id == trainee_id).first()
    if not t:
        raise HTTPException(404, "Trainee not found")
    return {
        "id": str(t.id),
        "admission_number": t.admission_number,
        "first_name": t.first_name,
        "last_name": t.last_name,
        "national_id": t.national_id,
        "phone": t.phone,
        "email": t.email,
        "status": t.status,
    }
