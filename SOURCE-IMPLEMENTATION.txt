# TVET e-ASSESS — Digital Practical Assessment Platform

A centralized web-based platform designed to digitize the entire practical assessment process for TVET institutions.

## Tech Stack
- **Frontend:** React.js + TypeScript + Tailwind CSS + shadcn/ui
- **Backend:** Python FastAPI
- **Database:** Supabase PostgreSQL
- **Storage:** Supabase Storage
- **Hosting:** Render

## Quick Start

### 1. Setup Supabase
1. Create a new Supabase project at https://supabase.com
2. Run `database/schema.sql` in the SQL Editor
3. Copy your project URL and keys

### 2. Backend
```bash
cd backend
python -m venv venv
source venv/bin/activate  # Windows: venv\Scripts\activate
pip install -r requirements.txt
cp .env.example .env  # Edit with your Supabase credentials
uvicorn app.main:app --reload --port 8000
3. Frontend
bash
cd frontend
npm install
cp .env.example .env.local  # Edit with your Supabase keys
npm run dev
4. Access
Frontend: http://localhost:5173
Backend API Docs: http://localhost:8000/api/docs
Project Motto
Assess Digitally. Verify Competence. Preserve Evidence.

-- ============================================================

CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. INSTITUTIONS
CREATE TABLE IF NOT EXISTS institutions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) UNIQUE NOT NULL,
    address TEXT,
    phone VARCHAR(50),
    email VARCHAR(100),
    logo_url TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. DEPARTMENTS
CREATE TABLE IF NOT EXISTS departments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL,
    hod_user_id UUID,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(institution_id, code)
);

-- 3. COURSES
CREATE TABLE IF NOT EXISTS courses (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    name VARCHAR(255) NOT NULL,
    code VARCHAR(50) NOT NULL,
    qualification_level VARCHAR(50),
    duration_months INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(department_id, code)
);

-- 4. CLASSES
CREATE TABLE IF NOT EXISTS classes (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    intake_year INTEGER,
    cycle INTEGER,
    level INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, name)
);

-- 5. UNITS
CREATE TABLE IF NOT EXISTS units (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    code VARCHAR(50) NOT NULL,
    name VARCHAR(255) NOT NULL,
    credit_hours INTEGER,
    level INTEGER,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(course_id, code)
);

-- 6. USERS
CREATE TABLE IF NOT EXISTS users (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    institution_id UUID REFERENCES institutions(id),
    department_id UUID REFERENCES departments(id),
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    phone VARCHAR(50),
    staff_number VARCHAR(50),
    role VARCHAR(50) NOT NULL CHECK (role IN (
        'super_admin', 'hod', 'trainer', 'assessor',
        'internal_verifier', 'external_verifier', 'trainee', 'management'
    )),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7. TRAINEES
CREATE TABLE IF NOT EXISTS trainees (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    admission_number VARCHAR(50) UNIQUE NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    national_id VARCHAR(50),
    phone VARCHAR(50),
    email VARCHAR(255),
    enrollment_date DATE,
    status VARCHAR(20) DEFAULT 'active' CHECK (status IN ('active', 'graduated', 'dropped', 'transferred')),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 8. TRAINER ALLOCATIONS
CREATE TABLE IF NOT EXISTS trainer_allocations (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    trainer_id UUID REFERENCES users(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    unit_id UUID REFERENCES units(id) ON DELETE CASCADE,
    academic_year INTEGER NOT NULL,
    term INTEGER NOT NULL CHECK (term IN (1, 2, 3)),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(trainer_id, class_id, unit_id, academic_year, term)
);

-- 9. ACADEMIC PERIODS
CREATE TABLE IF NOT EXISTS academic_periods (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    year INTEGER NOT NULL,
    term INTEGER NOT NULL CHECK (term IN (1, 2, 3)),
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(institution_id, year, term)
);

-- 10. ASSESSMENTS
CREATE TABLE IF NOT EXISTS assessments (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    institution_id UUID REFERENCES institutions(id) ON DELETE CASCADE,
    department_id UUID REFERENCES departments(id) ON DELETE CASCADE,
    course_id UUID REFERENCES courses(id) ON DELETE CASCADE,
    class_id UUID REFERENCES classes(id) ON DELETE CASCADE,
    unit_id UUID REFERENCES units(id) ON DELETE CASCADE,
    academic_period_id UUID REFERENCES academic_periods(id),
    title VARCHAR(255) NOT NULL,
    assessment_number VARCHAR(50) NOT NULL,
    assessment_type VARCHAR(20) NOT NULL CHECK (assessment_type IN ('oral', 'practical')),
    assessment_level VARCHAR(20),
    instructions TEXT,
    duration_minutes INTEGER,
    total_marks INTEGER,
    passing_percentage DECIMAL(5,2) DEFAULT 50.00,
    required_tools TEXT,
    required_equipment TEXT,
    required_materials TEXT,
    safety_precautions TEXT,
    ppe_requirements TEXT,
    marking_guidelines TEXT,
    reference_documents TEXT,
    status VARCHAR(20) DEFAULT 'draft' CHECK (status IN (
        'draft', 'preview', 'published', 'withdrawn', 'archived'
    )),
    version INTEGER DEFAULT 1,
    published_at TIMESTAMPTZ,
    deadline TIMESTAMPTZ,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(class_id, unit_id, assessment_number, academic_period_id, version)
);

-- 11. ASSESSMENT TASKS
CREATE TABLE IF NOT EXISTS assessment_tasks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    task_number INTEGER NOT NULL,
    task_title VARCHAR(255) NOT NULL,
    task_description TEXT,
    max_marks INTEGER NOT NULL DEFAULT 0,
    task_type VARCHAR(30) DEFAULT 'practical' CHECK (task_type IN (
        'practical', 'oral', 'theory', 'observation', 'continuous_test'
    )),
    display_order INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 12. CHECKLIST CRITERIA
CREATE TABLE IF NOT EXISTS checklist_criteria (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    task_id UUID REFERENCES assessment_tasks(id) ON DELETE CASCADE,
    criterion_number VARCHAR(20),
    criterion_text TEXT NOT NULL,
    sub_criterion_text TEXT,
    max_marks INTEGER NOT NULL DEFAULT 0,
    scoring_type VARCHAR(20) DEFAULT 'numerical' CHECK (scoring_type IN (
        'binary', 'numerical', 'award_max'
    )),
    award_max INTEGER,
    binary_pass_mark INTEGER DEFAULT 1,
    display_order INTEGER DEFAULT 0,
    is_sub_total BOOLEAN DEFAULT FALSE,
    is_grand_total BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 13. ASSESSMENT SESSIONS
CREATE TABLE IF NOT EXISTS assessment_sessions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    assessment_id UUID REFERENCES assessments(id) ON DELETE CASCADE,
    trainee_id UUID REFERENCES trainees(id) ON DELETE CASCADE,
    assessor_id UUID REFERENCES users(id),
    session_number INTEGER DEFAULT 1,
    assessment_date DATE,
    assessment_venue VARCHAR(255),
    academic_period_id UUID REFERENCES academic_periods(id),
    total_marks_available INTEGER DEFAULT 0,
    total_marks_obtained DECIMAL(10,2) DEFAULT 0,
    percentage_score DECIMAL(5,2) DEFAULT 0,
    competency_outcome VARCHAR(30) CHECK (competency_outcome IN (
        'competent', 'not_yet_competent', 'proficient', 'mastery', 'pending'
    )) DEFAULT 'pending',
    assessor_comments TEXT,
    feedback_to_candidate TEXT,
    status VARCHAR(20) DEFAULT 'in_progress' CHECK (status IN (
        'in_progress', 'submitted', 'verified', 'returned', 'rejected', 'approved'
    )),
    submitted_at TIMESTAMPTZ,
    verified_at TIMESTAMPTZ,
    approved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(assessment_id, trainee_id, session_number)
);

-- 14. CRITERION RESULTS
CREATE TABLE IF NOT EXISTS criterion_results (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    session_id UUID REFERENCES assessment_sessions(id) ON DELETE CASCADE,
    criterion_id UUID REFERENCES checklist_criteria(id) ON DELETE CASCADE,
    marks_available INTEGER NOT NULL,
    marks_obtained DECIMAL(10,2) DEFAULT 0,
    is_competent BOOLEAN,
    assessor_comment TEXT,
    evidence_attached BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(session_id, criterion_id)
);

-- 15. EVIDENCE FILES
CREATE TABLE IF NOT EXISTS evidence_files (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    session_id UUID REFERENCES assessment_sessions(id) ON DELETE CASCADE,

    admission_number VARCHAR(50),

    document_type VARCHAR(50),
    assessment_number VARCHAR(50),
    term INTEGER,
    year INTEGER,
    assessed_by_name VARCHAR(200),
    item_reference VARCHAR(50),
    version INTEGER DEFAULT 1,
    is_current BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMPTZ DEFAULT NOW()

CREATE TABLE IF NOT EXISTS generated_documents (

-- AUTO-UPDATE TRIGGER FUNCTION
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_institutions_updated BEFORE UPDATE ON institutions FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
CREATE TRIGGER trg_departments_updated BEFORE UPDATE ON departments FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

security = HTTPBearer()


async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db),
) -> User:
    token = credentials.credentials
    try:
        user_data = supabase.auth.get_user(token)
        if not user_data or not user_data.user:
            raise HTTPException(status_code=401, detail="Invalid token")
        user = db.query(User).filter(User.id == user_data.user.id).first()
        if not user:
            raise HTTPException(status_code=404, detail="User not found")
        if not user.is_active:
            raise HTTPException(status_code=403, detail="Account deactivated")
        return user
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=401, detail=f"Authentication failed: {str(e)}")


def require_role(*roles: str):
    def role_checker(current_user: User = Depends(get_current_user)) -> User:
        if current_user.role not in roles:
            raise HTTPException(status_code=403, detail=f"Required role(s): {', '.join(roles)}")
        return current_user
    return role_checker

from sqlalchemy.orm import Session
from app.database import get_db, supabase
from app.models.user import User
from app.auth.schemas import LoginRequest, LoginResponse, RegisterRequest
from app.auth.dependencies import get_current_user, require_admin

router = APIRouter()


@router.post("/login", response_model=LoginResponse)
def login(data: LoginRequest, db: Session = Depends(get_db)):
    try:
        auth_response = supabase.auth.sign_in_with_password({
            "email": data.email,
            "password": data.password,
        })
        if not auth_response.session:
            raise HTTPException(status_code=401, detail="Invalid credentials")

        user = db.query(User).filter(User.id == auth_response.user.id).first()
        if not user:
            raise HTTPException(status_code=404, detail="User profile not found")

        return LoginResponse(
            access_token=auth_response.session.access_token,
            user_id=str(user.id),
            role=user.role,
            full_name=f"{user.first_name} {user.last_name}",
        )
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=401, detail=str(e))


@router.post("/register")
def register(data: RegisterRequest, db: Session = Depends(get_db), admin: User = Depends(require_admin)):
    try:
        auth_response = supabase.auth.admin.create_user({
            "email": data.email,
            "password": data.password,
            "email_confirm": True,
            "user_metadata": {
                "first_name": data.first_name,
                "last_name": data.last_name,
            },
        })
        user = User(
            id=auth_response.user.id,
            email=data.email,
            first_name=data.first_name,
            last_name=data.last_name,
            role=data.role,
            department_id=data.department_id,
            staff_number=data.staff_number,
            phone=data.phone,
        )
        db.add(user)
        db.commit()
        db.refresh(user)
        return {"message": "User registered", "user_id": str(user.id)}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))


@router.get("/me")
def get_me(current_user: User = Depends(get_current_user)):
    return {
        "id": str(current_user.id),
        "email": current_user.email,
        "first_name": current_user.first_name,
        "last_name": current_user.last_name,
        "role": current_user.role,
        "department_id": str(current_user.department_id) if current_user.department_id else None,
    }


@router.post("/logout")

    supabase.auth.sign_out()
    return {"message": "Logged out"}

@router.get("/preview")
def preview_marksheet(
    class_id: uuid.UUID = Query(...),
    unit_id: uuid.UUID = Query(...),
    academic_period_id: uuid.UUID = Query(...),
    db: Session = Depends(get_db),
    user: User = Depends(require_trainer),
):

    from app.models.session import AssessmentSession
    from app.models.assessment import Assessment

    trainees = db.query(Trainee).filter(Trainee.class_id == class_id, Trainee.status == 'active').order_by(Trainee.admission_number).all()

        Assessment.class_id == class_id, Assessment.unit_id == unit_id,
        Assessment.academic_period_id == academic_period_id,
        Assessment.status.in_(['published', 'archived']),
    ).all()
    oral = [a for a in assessments if a.assessment_type == 'oral']
    practical = [a for a in assessments if a.assessment_type == 'practical']

    preview = []
    for t in trainees:
        oral_results = []
        for oa in oral:
            s = db.query(AssessmentSession).filter(AssessmentSession.assessment_id == oa.id, AssessmentSession.trainee_id == t.id, AssessmentSession.status.in_(['submitted', 'verified', 'approved'])).first()
            oral_results.append({"assessment": oa.assessment_number, "percentage": s.percentage_score if s else None})

        for pa in practical:
            s = db.query(AssessmentSession).filter(AssessmentSession.assessment_id == pa.id, AssessmentSession.trainee_id == t.id, AssessmentSession.status.in_(['submitted', 'verified', 'approved'])).first()
            prac_results.append({"assessment": pa.assessment_number, "percentage": s.percentage_score if s else None})

        oral_vals = [r['percentage'] for r in oral_results if r['percentage'] is not None]
        prac_vals = [r['percentage'] for r in prac_results if r['percentage'] is not None]
        oral_avg = sum(oral_vals) / len(oral_vals) if oral_vals else None
        prac_avg = sum(prac_vals) / len(prac_vals) if prac_vals else None
        overall = (oral_avg + prac_avg) / 2 if oral_avg is not None and prac_avg is not None else (oral_avg or prac_avg)

        preview.append({
            "admission_number": t.admission_number,
            "name": f"{t.first_name} {t.last_name}",
            "oral_results": oral_results,
            "practical_results": prac_results,
            "oral_average": round(oral_avg, 2) if oral_avg else None,
            "practical_average": round(prac_avg, 2) if prac_avg else None,
            "overall_average": round(overall, 2) if overall else None,
        })

    return {
        "oral_assessments": [a.assessment_number for a in oral],
        "practical_assessments": [a.assessment_number for a in practical],
        "trainees": preview,
    }

from sqlalchemy.orm import Session

import uuid
from app.models.assessment import Assessment, AssessmentTask, ChecklistCriterion
from app.models.audit import AuditLog


class AssessmentService:

    @staticmethod
    def create_assessment(db: Session, data: dict, created_by: uuid.UUID) -> Assessment:
        assessment = Assessment(**data, created_by=created_by, status='draft', version=1)
        db.add(assessment)
        db.flush()
        db.add(AuditLog(user_id=created_by, action="assessment_created", entity_type="assessment", entity_id=assessment.id, new_values={"title": data.get("title")}))
        db.commit()
        db.refresh(assessment)
        return assessment

    @staticmethod
    def add_task(db: Session, assessment_id: uuid.UUID, task_data: dict) -> AssessmentTask:
        count = db.query(AssessmentTask).filter(AssessmentTask.assessment_id == assessment_id).count()
        task = AssessmentTask(assessment_id=assessment_id, task_number=count + 1, **task_data)
        db.add(task)
        db.commit()
        db.refresh(task)
        return task

    @staticmethod
    def add_checklist_criterion(db: Session, assessment_id: uuid.UUID, task_id, criterion_data: dict) -> ChecklistCriterion:
        count = db.query(ChecklistCriterion).filter(ChecklistCriterion.assessment_id == assessment_id).count()
        criterion = ChecklistCriterion(assessment_id=assessment_id, task_id=task_id, criterion_number=str(count + 1), **criterion_data)
        db.add(criterion)
        db.commit()
        db.refresh(criterion)
        return criterion

    @staticmethod
    def publish_assessment(db: Session, assessment_id: uuid.UUID, user_id: uuid.UUID) -> Assessment:
        a = db.query(Assessment).filter(Assessment.id == assessment_id).first()
        if not a:
            raise ValueError("Assessment not found")
        if a.status != 'draft':
            raise ValueError("Only draft assessments can be published")
        from sqlalchemy import func
        total = db.query(func.coalesce(func.sum(ChecklistCriterion.max_marks), 0)).filter(ChecklistCriterion.assessment_id == assessment_id).scalar()
        a.status = 'published'
        a.published_at = datetime.utcnow()
        a.total_marks = total
        db.commit()
        db.refresh(a)
        return a

    @staticmethod
    def withdraw_assessment(db: Session, assessment_id: uuid.UUID, user_id: uuid.UUID) -> Assessment:
        a = db.query(Assessment).filter(Assessment.id == assessment_id).first()
        if not a:
            raise ValueError("Assessment not found")
        from app.models.session import AssessmentSession
        active = db.query(AssessmentSession).filter(AssessmentSession.assessment_id == assessment_id, AssessmentSession.status.in_(['in_progress', 'submitted'])).count()
        if active > 0:
            raise ValueError(f"Cannot withdraw: {active} active session(s)")
        a.status = 'withdrawn'
        db.commit()
        db.refresh(a)
        return a

    @staticmethod
    def create_new_version(db: Session, assessment_id: uuid.UUID, user_id: uuid.UUID) -> Assessment:
        original = db.query(Assessment).filter(Assessment.id == assessment_id).first()
        if not original:
            raise ValueError("Assessment not found")
        original.status = 'archived'
        new_a = Assessment(

            academic_period_id=original.academic_period_id, title=original.title,
            assessment_number=original.assessment_number, assessment_type=original.assessment_type,
            assessment_level=original.assessment_level, instructions=original.instructions,
            duration_minutes=original.duration_minutes, passing_percentage=original.passing_percentage,
            required_tools=original.required_tools, required_equipment=original.required_equipment,
            required_materials=original.required_materials, safety_precautions=original.safety_precautions,
            ppe_requirements=original.ppe_requirements, marking_guidelines=original.marking_guidelines,
            status='draft', version=original.version + 1, created_by=user_id,

        db.commit()

        return new_a

from sqlalchemy.orm import Session
from sqlalchemy import func
from datetime import datetime
import uuid
from app.models.session import AssessmentSession
from app.models.criterion import CriterionResult, ChecklistCriterion
from app.models.assessment import Assessment
from app.models.audit import AuditLog


class MarkingService:

    @staticmethod
    def create_assessment_session(db: Session, assessment_id: uuid.UUID, trainee_id: uuid.UUID, assessor_id: uuid.UUID, assessment_date, venue=None, session_number=1) -> AssessmentSession:
        existing = db.query(AssessmentSession).filter(AssessmentSession.assessment_id == assessment_id, AssessmentSession.trainee_id == trainee_id, AssessmentSession.session_number == session_number).first()
        if existing and existing.status in ['submitted', 'verified', 'approved']:
            raise ValueError(f"Session already {existing.status}")
        if existing:
            return existing
        a = db.query(Assessment).filter(Assessment.id == assessment_id).first()
        if not a:
            raise ValueError("Assessment not found")
        total = db.query(func.coalesce(func.sum(ChecklistCriterion.max_marks), 0)).filter(ChecklistCriterion.assessment_id == assessment_id).scalar()
        s = AssessmentSession(assessment_id=assessment_id, trainee_id=trainee_id, assessor_id=assessor_id, session_number=session_number, assessment_date=assessment_date, assessment_venue=venue, academic_period_id=a.academic_period_id, total_marks_available=total, status='in_progress')
        db.add(s)
        db.commit()
        db.refresh(s)
        return s

    @staticmethod
    def record_criterion_mark(db: Session, session_id: uuid.UUID, criterion_id: uuid.UUID, marks_obtained: float, assessor_comment=None, user_id=None) -> CriterionResult:
        c = db.query(ChecklistCriterion).filter(ChecklistCriterion.id == criterion_id).first()
        if not c:
            raise ValueError("Criterion not found")
        if marks_obtained < 0 or marks_obtained > c.max_marks:
            raise ValueError(f"Marks must be between 0 and {c.max_marks}")
        is_competent = marks_obtained >= c.binary_pass_mark if c.scoring_type == 'binary' else None
        r = db.query(CriterionResult).filter(CriterionResult.session_id == session_id, CriterionResult.criterion_id == criterion_id).first()
        if r:
            r.marks_obtained = marks_obtained
            r.assessor_comment = assessor_comment
            r.is_competent = is_competent
        else:
            r = CriterionResult(session_id=session_id, criterion_id=criterion_id, marks_available=c.max_marks, marks_obtained=marks_obtained, assessor_comment=assessor_comment, is_competent=is_competent)
            db.add(r)
        db.commit()
        db.refresh(r)
        if user_id:
            db.add(AuditLog(user_id=user_id, action="criterion_mark_recorded", entity_type="criterion_result", entity_id=r.id, new_values={"marks_obtained": marks_obtained}))
            db.commit()
        return r

    @staticmethod
    def calculate_session_totals(db: Session, session_id: uuid.UUID) -> dict:
        s = db.query(AssessmentSession).filter(AssessmentSession.id == session_id).first()
        if not s:
            raise ValueError("Session not found")
        total_obtained = db.query(func.coalesce(func.sum(CriterionResult.marks_obtained), 0)).filter(CriterionResult.session_id == session_id).scalar()
        total_available = s.total_marks_available
        percentage = (total_obtained / total_available * 100) if total_available > 0 else 0
        a = db.query(Assessment).filter(Assessment.id == s.assessment_id).first()
        passing = a.passing_percentage if a else 50.0
        if percentage >= 80:
            competency = 'mastery'
        elif percentage >= 70:
            competency = 'proficient'
        elif percentage >= passing:
            competency = 'competent'
        else:
            competency = 'not_yet_competent'
        s.total_marks_obtained = round(total_obtained, 2)
        s.percentage_score = round(percentage, 2)
        s.competency_outcome = competency
        db.commit()
        db.refresh(s)
        return {"session_id": str(s.id), "total_obtained": total_obtained, "total_available": total_available, "percentage": round(percentage, 2), "competency": competency}

    @staticmethod
    def submit_session(db: Session, session_id: uuid.UUID, user_id: uuid.UUID, assessor_comments=None, feedback_to_candidate=None) -> AssessmentSession:
        s = db.query(AssessmentSession).filter(AssessmentSession.id == session_id).first()
        if not s:
            raise ValueError("Session not found")
        if s.status != 'in_progress':
            raise ValueError(f"Cannot submit: status is {s.status}")
        MarkingService.calculate_session_totals(db, session_id)
        s.assessor_comments = assessor_comments
        s.feedback_to_candidate = feedback_to_candidate
        s.status = 'submitted'
        s.submitted_at = datetime.utcnow()
        db.commit()
        db.refresh(s)
        db.add(AuditLog(user_id=user_id, action="session_submitted", entity_type="assessment_session", entity_id=session_id, new_values={"total_marks_obtained": s.total_marks_obtained, "percentage": s.percentage_score, "competency": s.competency_outcome}))
        db.commit()
        return s

    @staticmethod
    def get_session_summary(db: Session, session_id: uuid.UUID) -> dict:
        s = db.query(AssessmentSession).filter(AssessmentSession.id == session_id).first()
        if not s:
            raise ValueError("Session not found")
        results = db.query(CriterionResult).filter(CriterionResult.session_id == session_id).order_by(CriterionResult.created_at).all()
        criteria_details = []
        for cr in results:
            c = db.query(ChecklistCriterion).filter(ChecklistCriterion.id == cr.criterion_id).first()
            criteria_details.append({
                "criterion_id": str(cr.criterion_id),
                "criterion_text": c.criterion_text if c else "",
                "sub_criterion_text": c.sub_criterion_text if c else "",
                "max_marks": cr.marks_available,
                "marks_obtained": cr.marks_obtained,
                "is_competent": cr.is_competent,
                "comment": cr.assessor_comment,
                "scoring_type": c.scoring_type if c else "numerical",
            })
        return {
            "session_id": str(s.id),
            "assessment_id": str(s.assessment_id),
            "trainee_id": str(s.trainee_id),
            "assessor_id": str(s.assessor_id),
            "session_number": s.session_number,
            "assessment_date": s.assessment_date.isoformat() if s.assessment_date else None,
            "venue": s.assessment_venue,
            "total_marks_available": s.total_marks_available,
            "total_marks_obtained": s.total_marks_obtained,
            "percentage_score": s.percentage_score,
            "competency_outcome": s.competency_outcome,
            "assessor_comments": s.assessor_comments,
            "feedback_to_candidate": s.feedback_to_candidate,
            "status": s.status,
            "criteria_results": criteria_details,
        }

class MarksheetService:
    def __init__(self, db: Session):
        self.db = db

    def generate_marksheet(self, class_id: uuid.UUID, unit_id: uuid.UUID, academic_period_id: uuid.UUID, output_format: str = 'excel') -> dict:
        class_obj = self.db.query(Class).filter(Class.id == class_id).first()
        unit = self.db.query(Unit).filter(Unit.id == unit_id).first()
        trainees = self.db.query(Trainee).filter(Trainee.class_id == class_id, Trainee.status == 'active').order_by(Trainee.admission_number).all()
        assessments = self.db.query(Assessment).filter(Assessment.class_id == class_id, Assessment.unit_id == unit_id, Assessment.academic_period_id == academic_period_id, Assessment.status.in_(['published', 'archived'])).order_by(Assessment.assessment_type, Assessment.assessment_number).all()
        oral = [a for a in assessments if a.assessment_type == 'oral']
        practical = [a for a in assessments if a.assessment_type == 'practical']

        data = []
        for t in trainees:
            row = {"admission_number": t.admission_number, "trainee_name": f"{t.first_name} {t.last_name}", "oral_marks": {}, "practical_marks": {}}
            oral_total, oral_count = 0, 0
            for oa in oral:
                s = self.db.query(AssessmentSession).filter(AssessmentSession.assessment_id == oa.id, AssessmentSession.trainee_id == t.id, AssessmentSession.status.in_(['submitted', 'verified', 'approved'])).first()
                if s:
                    row["oral_marks"][oa.assessment_number] = {"marks_obtained": s.total_marks_obtained, "marks_available": s.total_marks_available, "percentage": s.percentage_score, "competency": s.competency_outcome}
                    oral_total += s.percentage_score
                    oral_count += 1
                else:
                    row["oral_marks"][oa.assessment_number] = None
            row["oral_average"] = (oral_total / oral_count) if oral_count > 0 else None

            prac_total, prac_count = 0, 0
            for pa in practical:
                s = self.db.query(AssessmentSession).filter(AssessmentSession.assessment_id == pa.id, AssessmentSession.trainee_id == t.id, AssessmentSession.status.in_(['submitted', 'verified', 'approved'])).first()
                if s:
                    row["practical_marks"][pa.assessment_number] = {"marks_obtained": s.total_marks_obtained, "marks_available": s.total_marks_available, "percentage": s.percentage_score, "competency": s.competency_outcome}
                    prac_total += s.percentage_score
                    prac_count += 1
                else:
                    row["practical_marks"][pa.assessment_number] = None
            row["practical_average"] = (prac_total / prac_count) if prac_count > 0 else None

            if row["oral_average"] is not None and row["practical_average"] is not None:
                row["overall_average"] = (row["oral_average"] + row["practical_average"]) / 2
            else:
                row["overall_average"] = row["oral_average"] or row["practical_average"]
            data.append(row)

        if output_format == 'excel':
            return self._generate_excel(data, oral, practical, class_obj, unit)
        return self._generate_csv(data, oral, practical, class_obj, unit)

    def _generate_excel(self, data, oral, practical, class_obj, unit) -> dict:
        wb = openpyxl.Workbook()
        ws = wb.active
        ws.title = "Marksheet"
        header_font = Font(name='Calibri', bold=True, size=11, color='FFFFFF')
        header_fill = PatternFill(start_color='2C3E50', end_color='2C3E50', fill_type='solid')
        avg_fill = PatternFill(start_color='27AE60', end_color='27AE60', fill_type='solid')
        avg_font = Font(name='Calibri', bold=True, size=10, color='FFFFFF')
        border = Border(left=Side(style='thin'), right=Side(style='thin'), top=Side(style='thin'), bottom=Side(style='thin'))
        center = Alignment(horizontal='center', vertical='center', wrap_text=True)

        ws.merge_cells('A1:Z1')
        ws['A1'] = "TVET e-ASSESS — CONSOLIDATED MARKSHEET"
        ws['A1'].font = Font(name='Calibri', bold=True, size=14)
        ws['A1'].alignment = Alignment(horizontal='center')

        row = 5
        col = 1
        for h in ['ADM NO.', 'TRAINEE NAME']:
            c = ws.cell(row=row, column=col, value=h)
            c.font = header_font; c.fill = header_fill; c.alignment = center; c.border = border
            col += 1
        for oa in oral:
            c = ws.cell(row=row, column=col, value=f"ORAL {oa.assessment_number}")
            c.font = header_font; c.fill = PatternFill(start_color='E67E22', end_color='E67E22', fill_type='solid'); c.alignment = center; c.border = border
            col += 1
        c = ws.cell(row=row, column=col, value="ORAL AVG."); c.font = avg_font; c.fill = avg_fill; c.alignment = center; c.border = border; col += 1
        for pa in practical:
            c = ws.cell(row=row, column=col, value=f"PRAC. {pa.assessment_number}")
            c.font = header_font; c.fill = PatternFill(start_color='2980B9', end_color='2980B9', fill_type='solid'); c.alignment = center; c.border = border
            col += 1
        c = ws.cell(row=row, column=col, value="PRAC. AVG."); c.font = avg_font; c.fill = avg_fill; c.alignment = center; c.border = border; col += 1
        c = ws.cell(row=row, column=col, value="OVERALL AVG."); c.font = avg_font; c.fill = PatternFill(start_color='8E44AD', end_color='8E44AD', fill_type='solid'); c.alignment = center; c.border = border; col += 1
        c = ws.cell(row=row, column=col, value="COMPETENCY"); c.font = header_font; c.fill = header_fill; c.alignment = center; c.border = border

        for i, td in enumerate(data):
            row = 6 + i; col = 1
            for val in [td['admission_number'], td['trainee_name']]:
                c = ws.cell(row=row, column=col, value=val); c.font = Font(name='Calibri', size=10); c.alignment = center if col == 1 else Alignment(horizontal='left'); c.border = border; col += 1
            for oa in oral:
                r = td['oral_marks'].get(oa.assessment_number)
                c = ws.cell(row=row, column=col, value=r['percentage'] if r else "-"); c.font = Font(name='Calibri', size=10); c.alignment = center; c.border = border; col += 1
            c = ws.cell(row=row, column=col, value=td['oral_average'] if td['oral_average'] is not None else "-"); c.font = Font(name='Calibri', bold=True, size=10); c.alignment = center; c.border = border; col += 1
            for pa in practical:
                r = td['practical_marks'].get(pa.assessment_number)
                c = ws.cell(row=row, column=col, value=r['percentage'] if r else "-"); c.font = Font(name='Calibri', size=10); c.alignment = center; c.border = border; col += 1
            c = ws.cell(row=row, column=col, value=td['practical_average'] if td['practical_average'] is not None else "-"); c.font = Font(name='Calibri', bold=True, size=10); c.alignment = center; c.border = border; col += 1
            c = ws.cell(row=row, column=col, value=td['overall_average'] if td['overall_average'] is not None else "-"); c.font = Font(name='Calibri', bold=True, size=10); c.alignment = center; c.border = border; col += 1
            overall = td['overall_average']
            comp = "Competent" if overall is not None and overall >= 50 else ("Not Yet Competent" if overall is not None else "Pending")
            c = ws.cell(row=row, column=col, value=comp); c.font = Font(name='Calibri', bold=True, size=10); c.alignment = center; c.border = border
            if comp == "Competent":
                c.fill = PatternFill(start_color='D5F5E3', end_color='D5F5E3', fill_type='solid')
            elif comp == "Not Yet Competent":
                c.fill = PatternFill(start_color='FADBD8', end_color='FADBD8', fill_type='solid')

        ws.column_dimensions['A'].width = 15
        ws.column_dimensions['B'].width = 30
        for cl in [get_column_letter(i) for i in range(3, col + 1)]:
            ws.column_dimensions[cl].width = 14

        output_dir = tempfile.gettempdir()
        filename = FilenameService.generate_marksheet_filename(unit.code if unit else "UNKNOWN", class_obj.name if class_obj else "UNKNOWN", "TERM1-2026")
        filepath = os.path.join(output_dir, filename)
        wb.save(filepath)
        return {"filepath": filepath, "filename": filename, "file_size_bytes": os.path.getsize(filepath), "trainee_count": len(data)}

    def _generate_csv(self, data, oral, practical, class_obj, unit) -> dict:
        output = io.StringIO()
        writer = csv.writer(output)
        header = ['ADM NO.', 'TRAINEE NAME']
        for oa in oral: header.append(f'ORAL {oa.assessment_number}')
        header.append('ORAL AVG.')
        for pa in practical: header.append(f'PRAC. {pa.assessment_number}')
        header.append('PRAC. AVG.'); header.append('OVERALL AVG.'); header.append('COMPETENCY')
        writer.writerow(header)
        for td in data:
            row = [td['admission_number'], td['trainee_name']]
            for oa in oral:
                r = td['oral_marks'].get(oa.assessment_number)
                row.append(f"{r['percentage']:.2f}" if r else "-")
            row.append(f"{td['oral_average']:.2f}" if td['oral_average'] is not None else "-")
            for pa in practical:
                r = td['practical_marks'].get(pa.assessment_number)
                row.append(f"{r['percentage']:.2f}" if r else "-")
            row.append(f"{td['practical_average']:.2f}" if td['practical_average'] is not None else "-")
            row.append(f"{td['overall_average']:.2f}" if td['overall_average'] is not None else "-")
            overall = td['overall_average']
            comp = "Competent" if overall is not None and overall >= 50 else ("Not Yet Competent" if overall is not None else "Pending")
            row.append(comp)
            writer.writerow(row)
        output_dir = tempfile.gettempdir()
        filename = f"marksheet_{unit.code if unit else 'unknown'}_{class_obj.name if class_obj else 'unknown'}.csv"
        filepath = os.path.join(output_dir, filename)
        with open(filepath, 'w', newline='') as f:
            f.write(output.getvalue())
        return {"filepath": filepath, "filename": filename, "file_size_bytes": os.path.getsize(filepath)}

        return {"filepath": filepath, "filename": filename, "file_size_bytes": os.path.getsize(filepath), "session_id": str(session_id)}

from sqlalchemy.orm import Session
import uuid
import mimetypes
import os
from supabase import create_client
from app.config import settings
from app.models.evidence import EvidenceFile
from app.models.session import AssessmentSession
from app.models.trainee import Trainee
from app.models.assessment import Assessment

supabase = create_client(settings.SUPABASE_URL, settings.SUPABASE_SERVICE_ROLE_KEY)


class EvidenceService:
    ALLOWED_TYPES = {
        'photo': settings.ALLOWED_IMAGE_TYPES,
        'video': settings.ALLOWED_VIDEO_TYPES,
        'audio': settings.ALLOWED_AUDIO_TYPES,
        'document': settings.ALLOWED_DOC_TYPES,
        'scanned': settings.ALLOWED_IMAGE_TYPES + ['application/pdf'],
    }
    MAX_FILE_SIZE = settings.MAX_FILE_SIZE_MB * 1024 * 1024

    @staticmethod
    def _detect_file_type(mime_type: str) -> str:
        for category, types in EvidenceService.ALLOWED_TYPES.items():
            if mime_type in types:
                return category
        return 'document'

    @classmethod
    def upload_evidence(cls, db: Session, session_id: uuid.UUID, criterion_id, file_content: bytes, original_filename: str, uploaded_by: uuid.UUID, item_reference=None) -> EvidenceFile:
        s = db.query(AssessmentSession).filter(AssessmentSession.id == session_id).first()
        if not s:
            raise ValueError("Session not found")
        t = db.query(Trainee).filter(Trainee.id == s.trainee_id).first()
        a = db.query(Assessment).filter(Assessment.id == s.assessment_id).first()
        assessor = db.query(User).filter(User.id == uploaded_by).first()

        mime_type, _ = mimetypes.guess_type(original_filename)
        if not mime_type:
            mime_type = 'application/octet-stream'
        file_type = cls._detect_file_type(mime_type)

        if len(file_content) > cls.MAX_FILE_SIZE:
            raise ValueError(f"File exceeds {settings.MAX_FILE_SIZE_MB}MB")
        if mime_type not in cls.ALLOWED_TYPES.get(file_type, []):
            raise ValueError(f"File type {mime_type} not allowed")

        ext = original_filename.rsplit('.', 1)[-1] if '.' in original_filename else 'bin'
        existing = db.query(EvidenceFile).filter(EvidenceFile.session_id == session_id, EvidenceFile.criterion_id == criterion_id, EvidenceFile.is_current == True).all()
        file_index = len(existing) + 1

        system_filename = FilenameService.generate_evidence_filename(
            t.admission_number, a.unit.code if a.unit else "UNKNOWN", file_type.upper(),
            a.assessment_number, a.academic_period.term if a.academic_period else 1,
            a.academic_period.year if a.academic_period else 2026,
            f"{assessor.first_name}_{assessor.last_name}" if assessor else "Unknown",
            item_reference, file_index, ext
        )
        storage_path = f"evidence/{session_id}/{system_filename}"

        try:
            supabase.storage.from_(settings.SUPABASE_STORAGE_BUCKET).upload(path=storage_path, file=file_content, file_options={"content-type": mime_type})
        except Exception as e:
            raise ValueError(f"Upload failed: {str(e)}")

        for ef in existing:
            ef.is_current = False
            ef.version += 1

        evidence = EvidenceFile(
            session_id=session_id, criterion_id=criterion_id, uploaded_by=uploaded_by,
            original_filename=original_filename, system_filename=system_filename, storage_path=storage_path,
            file_type=file_type, mime_type=mime_type, file_size_bytes=len(file_content),
            admission_number=t.admission_number, unit_code=a.unit.code if a.unit else None,
            document_type=file_type.upper(), assessment_number=a.assessment_number,
            term=a.academic_period.term if a.academic_period else None,
            year=a.academic_period.year if a.academic_period else None,

from sqlalchemy.orm import Session
from datetime import datetime
import uuid
from app.models.session import AssessmentSession
from app.models.verification import VerificationRecord
from app.models.user import User
from app.models.audit import AuditLog


class VerificationService:

    @staticmethod
    def verify_assessment(db: Session, session_id: uuid.UUID, verifier_id: uuid.UUID, verifier_type: str, decision: str, comments=None, rejection_reasons=None) -> VerificationRecord:
        s = db.query(AssessmentSession).filter(AssessmentSession.id == session_id).first()
        if not s:
            raise ValueError("Session not found")
        if s.assessor_id == verifier_id:
            raise ValueError("Assessors cannot verify their own assessments")
        v = db.query(User).filter(User.id == verifier_id).first()
        if verifier_type == 'internal' and v.role not in ['super_admin', 'internal_verifier', 'hod']:
            raise ValueError("Not authorized as internal verifier")
        if verifier_type == 'external' and v.role != 'external_verifier':
            raise ValueError("Not authorized as external verifier")
        if s.status not in ['submitted', 'returned']:
            raise ValueError(f"Cannot verify: status is {s.status}")

        record = VerificationRecord(session_id=session_id, verifier_id=verifier_id, verifier_type=verifier_type, decision=decision, comments=comments, rejection_reasons=rejection_reasons, verified_at=datetime.utcnow())
        db.add(record)

        if decision == 'approved':
            s.status = 'approved'; s.approved_at = datetime.utcnow()
        elif decision == 'returned_for_correction':
            s.status = 'returned'
        elif decision == 'rejected':
            s.status = 'rejected'

        db.commit()
        db.refresh(record)
        db.add(AuditLog(user_id=verifier_id, action=f"assessment_{decision}", entity_type="verification", entity_id=record.id, new_values={"session_id": str(session_id), "decision": decision}))
        db.commit()
        return record

    @staticmethod
    def get_verification_history(db: Session, session_id: uuid.UUID):
        records = db.query(VerificationRecord).filter(VerificationRecord.session_id == session_id).order_by(VerificationRecord.verified_at.desc()).all()
        result = []
        for r in records:
            v = db.query(User).filter(User.id == r.verifier_id).first()
            result.append({
                "verification_id": str(r.id),
                "verifier_name": f"{v.first_name} {v.last_name}" if v else "Unknown",
                "verifier_type": r.verifier_type,
                "decision": r.decision,
                "comments": r.comments,
                "rejection_reasons": r.rejection_reasons,
                "verified_at": r.verified_at.isoformat() if r.verified_at else None,
            })
        return result

import re
from typing import Optional


class FilenameService:

    @staticmethod
    def _sanitize(text: str) -> str:
        text = re.sub(r'[^\w\s-]', '', text)
        text = re.sub(r'[\s]+', '_', text.strip())
        return text.upper()

    @classmethod
    def generate_assessment_filename(cls, admission_number: str, unit_name: str, document_type: str, assessment_number: str, term: int, year: int, assessed_by_name: str, item_reference: Optional[str] = None, file_index: Optional[int] = None, extension: str = "pdf") -> str:
        parts = [cls._sanitize(admission_number), cls._sanitize(unit_name), cls._sanitize(document_type), cls._sanitize(assessment_number), f"TERM{term}", str(year), "ASSESSED_BY", cls._sanitize(assessed_by_name)]
        if item_reference:
            parts.append(cls._sanitize(item_reference))
        if file_index is not None:
            parts.append(f"FILE{file_index:02d}")
        return "-".join(parts) + f".{extension}"

    @classmethod
    def generate_evidence_filename(cls, admission_number: str, unit_code: str, document_type: str, assessment_number: str, term: int, year: int, assessed_by_name: str, item_reference: Optional[str] = None, file_index: int = 1, extension: str = "jpg") -> str:
        return cls.generate_assessment_filename(admission_number, unit_code, document_type, assessment_number, term, year, assessed_by_name, item_reference, file_index, extension)

    @classmethod
    def generate_marksheet_filename(cls, unit_code: str, class_name: str, academic_period: str, extension: str = "xlsx") -> str:
        return f"MARKSHEET-{cls._sanitize(unit_code)}-{cls._sanitize(class_name)}-{cls._sanitize(academic_period)}.{extension}"

    @classmethod
    def generate_report_filename(cls, report_type: str, department: str, academic_period: str, extension: str = "pdf") -> str:
        return f"REPORT-{cls._sanitize(report_type)}-{cls._sanitize(department)}-{cls._sanitize(academic_period)}.{extension}"

  user: null,
  loading: true,
  setUser: (user) => set({ user }),
  setLoading: (loading) => set({ loading }),
  logout: async () => {

    set({ user: null });

    const { data: { session } } = await supabase.auth.getSession();
    if (session) {

        const res = await fetch(`${import.meta.env.VITE_API_BASE_URL || ''}/api/v1/auth/me`, {
          headers: { Authorization: `Bearer ${session.access_token}` },
        });
        if (res.ok) {
          const user = await res.json();
          set({ user, loading: false });
        } else {
          set({ loading: false });
        }
      } catch {
        set({ loading: false });
      }
    } else {
      set({ loading: false });
    }
  },
}));

import { useState, useEffect } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from '../ui/card';
import { Button } from '../ui/button';
import { Input } from '../ui/input';
import { Textarea } from '../ui/textarea';
import { Badge } from '../ui/badge';
import { Progress } from '../ui/progress';
import { Alert, AlertDescription } from '../ui/alert';
import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '../ui/table';
import { CheckCircle, Save, Send } from 'lucide-react';
import { sessionsApi } from '../../api/sessions';
import { markingApi } from '../../api/marking';
import { EvidenceUploader } from '../evidence/EvidenceUploader';
import { COMPETENCY_COLORS } from '../../utils/constants';
import type { ChecklistCriterion } from '../../types/assessment';

export function MarkingInterface({ sessionId }: { sessionId: string }) {
  const [session, setSession] = useState<any>(null);
  const [criteria, setCriteria] = useState<ChecklistCriterion[]>([]);
  const [results, setResults] = useState<Record<string, any>>({});
  const [comments, setComments] = useState<Record<string, string>>({});
  const [assessorComments, setAssessorComments] = useState('');
  const [feedback, setFeedback] = useState('');
  const [error, setError] = useState<string | null>(null);
  const [success, setSuccess] = useState<string | null>(null);
  const [tab, setTab] = useState<'checklist' | 'evidence' | 'comments'>('checklist');

  useEffect(() => { loadSession(); }, [sessionId]);

  const loadSession = async () => {
    try {
      const res = await sessionsApi.getSummary(sessionId);
      setSession(res.data);
      setCriteria(res.data.criteria_results || []);
      const resultsMap: Record<string, any> = {};
      (res.data.criteria_results || []).forEach((cr: any) => { resultsMap[cr.criterion_id] = cr; if (cr.comment) setComments((p) => ({ ...p, [cr.criterion_id]: cr.comment })); });
      setResults(resultsMap);
      setAssessorComments(res.data.assessor_comments || '');
      setFeedback(res.data.feedback_to_candidate || '');
    } catch (err: any) { setError(err.response?.data?.detail || 'Failed to load'); }
  };

  const handleMarkChange = (criterionId: string, value: number, maxMarks: number) => {
    if (value < 0) value = 0;
    if (value > maxMarks) value = maxMarks;
    setResults((p) => ({ ...p, [criterionId]: { ...(p[criterionId] || {}), marks_obtained: value, marks_available: maxMarks } }));
  };

  const saveMarks = async () => {
    const marks = Object.entries(results).map(([cid, r]: [string, any]) => ({ criterion_id: cid, marks_obtained: r.marks_obtained || 0, assessor_comment: comments[cid] || undefined }));
    try {
      await markingApi.recordMarks(sessionId, marks);
      setSuccess('Marks saved');
      setTimeout(() => setSuccess(null), 3000);
    } catch (err: any) { setError(err.response?.data?.detail || 'Failed to save'); }
  };

  const submit = async () => {
    if (!window.confirm('Submit for verification?')) return;
    try {
      await markingApi.submit(sessionId, { assessor_comments: assessorComments, feedback_to_candidate: feedback });
      setSuccess('Submitted for verification');
    } catch (err: any) { setError(err.response?.data?.detail || 'Failed to submit'); }
  };

  if (!session) return <div className="p-8 text-center">Loading...</div>;

  const totalObtained = Object.values(results).reduce((s: number, r: any) => s + (r.marks_obtained || 0), 0);
  const totalAvailable = session.total_marks_available || 0;
  const percentage = totalAvailable > 0 ? (totalObtained / totalAvailable) * 100 : 0;

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold">Assessment Session</h1>
          <p className="text-muted-foreground">Session #{session.session_number}</p>
        </div>
        <Badge className={COMPETENCY_COLORS[session.competency_outcome] || ''}>{session.competency_outcome?.replace('_', ' ').toUpperCase()}</Badge>
      </div>

      <Card>
        <CardContent className="pt-6">
          <div className="grid grid-cols-4 gap-4">
            <div className="text-center"><p className="text-sm text-muted-foreground">Obtained</p><p className="text-2xl font-bold">{totalObtained.toFixed(1)}</p></div>
            <div className="text-center"><p className="text-sm text-muted-foreground">Available</p><p className="text-2xl font-bold">{totalAvailable}</p></div>
            <div className="text-center"><p className="text-sm text-muted-foreground">Percentage</p><p className="text-2xl font-bold">{percentage.toFixed(1)}%</p></div>
            <div className="text-center"><p className="text-sm text-muted-foreground">Pass Mark</p><p className="text-2xl font-bold">{session.passing_percentage || 50}%</p></div>
          </div>
          <Progress value={percentage} className="mt-4" />
        </CardContent>
      </Card>

      {success && <Alert className="bg-green-50 border-green-200"><CheckCircle className="h-4 w-4 text-green-600" /><AlertDescription className="text-green-800">{success}</AlertDescription></Alert>}
      {error && <Alert variant="destructive"><AlertDescription>{error}</AlertDescription></Alert>}

      <div className="flex gap-2 border-b">
        {(['checklist', 'evidence', 'comments'] as const).map((t) => (
          <button key={t} onClick={() => setTab(t)} className={`px-4 py-2 text-sm font-medium ${tab === t ? 'border-b-2 border-primary' : 'text-muted-foreground'}`}>{t.charAt(0).toUpperCase() + t.slice(1)}</button>
        ))}
      </div>

      {tab === 'checklist' && (
        <Card>
          <CardHeader><CardTitle>Observation Checklist</CardTitle></CardHeader>
          <CardContent>
            <Table>
              <TableHeader><TableRow><TableHead className="w-[40%]">Item</TableHead><TableHead className="w-[15%] text-center">Available</TableHead><TableHead className="w-[20%] text-center">Obtained</TableHead><TableHead className="w-[25%]">Comments</TableHead></TableRow></TableHeader>
              <TableBody>
                {criteria.map((c: any) => (
                  <TableRow key={c.criterion_id}>
                    <TableCell><p className="font-medium">{c.criterion_text}</p>{c.sub_criterion_text && <p className="text-sm text-muted-foreground italic">({c.sub_criterion_text})</p>}</TableCell>
                    <TableCell className="text-center font-medium">{c.max_marks}</TableCell>
                    <TableCell className="text-center"><Input type="number" min={0} max={c.max_marks} value={results[c.criterion_id]?.marks_obtained || 0} onChange={(e) => handleMarkChange(c.criterion_id, parseFloat(e.target.value) || 0, c.max_marks)} className="w-20 mx-auto text-center" /></TableCell>
                    <TableCell><Input placeholder="Comment" value={comments[c.criterion_id] || ''} onChange={(e) => setComments((p) => ({ ...p, [c.criterion_id]: e.target.value }))} /></TableCell>
                  </TableRow>
                ))}
              </TableBody>
            </Table>
          </CardContent>
        </Card>
      )}

      {tab === 'evidence' && <Card><CardHeader><CardTitle>Multimedia Evidence</CardTitle></CardHeader><CardContent><EvidenceUploader sessionId={sessionId} criteria={criteria} /></CardContent></Card>}

      {tab === 'comments' && (
        <Card>
          <CardHeader><CardTitle>Comments & Submit</CardTitle></CardHeader>
          <CardContent className="space-y-4">
            <div><label className="text-sm font-medium">Assessor Comments</label><Textarea value={assessorComments} onChange={(e) => setAssessorComments(e.target.value)} rows={4} /></div>
            <div><label className="text-sm font-medium">Feedback to Candidate</label><Textarea value={feedback} onChange={(e) => setFeedback(e.target.value)} rows={4} /></div>
            <div className="flex gap-4">
              <Button onClick={saveMarks} variant="outline"><Save className="mr-2 h-4 w-4" />Save Progress</Button>
              <Button onClick={submit} className="bg-green-600 hover:bg-green-700"><Send className="mr-2 h-4 w-4" />Submit for Verification</Button>
            </div>
          </CardContent>
        </Card>
      )}
    </div>
  );
}

import { useState, useRef } from 'react';
import { Card, CardContent } from '../ui/card';
import { Button } from '../ui/button';
import { Badge } from '../ui/badge';
import { Progress } from '../ui/progress';
import { Upload, FileImage, FileVideo, FileAudio, FileText, CheckCircle } from 'lucide-react';
import { evidenceApi } from '../../api/evidence';
import type { ChecklistCriterion } from '../../types/assessment';

export function EvidenceUploader({ sessionId, criteria }: { sessionId: string; criteria: ChecklistCriterion[] }) {
  const [selectedCriterion, setSelectedCriterion] = useState('');
  const [uploading, setUploading] = useState(false);
  const [progress, setProgress] = useState(0);
  const [files, setFiles] = useState<any[]>([]);
  const [error, setError] = useState<string | null>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const getFileIcon = (type: string) => {
    switch (type) {
      case 'photo': return <FileImage className="h-5 w-5 text-blue-500" />;
      case 'video': return <FileVideo className="h-5 w-5 text-purple-500" />;
      case 'audio': return <FileAudio className="h-5 w-5 text-orange-500" />;
      default: return <FileText className="h-5 w-5 text-gray-500" />;
    }
  };

  const handleFileSelect = async (event: React.ChangeEvent<HTMLInputElement>) => {
    const fileList = event.target.files;
    if (!fileList || fileList.length === 0) return;
    if (!selectedCriterion) { setError('Select a criterion first'); return; }
    setUploading(true); setProgress(0); setError(null);
    for (const file of Array.from(fileList)) {
      try {
        const res = await evidenceApi.upload(sessionId, selectedCriterion, file);
        setFiles((p) => [...p, res.data]);
        setProgress(100);
      } catch (err: any) { setError(err.response?.data?.detail || `Failed: ${file.name}`); }
    }
    setUploading(false);
    if (fileInputRef.current) fileInputRef.current.value = '';
  };

  return (
    <div className="space-y-4">
      <div>
        <label className="text-sm font-medium mb-2 block">Select Criterion</label>
        <select value={selectedCriterion} onChange={(e) => setSelectedCriterion(e.target.value)} className="w-full p-2 border rounded-md">
          <option value="">-- Select --</option>
          {criteria.filter((c) => !c.is_sub_total && !c.is_grand_total).map((c) => (<option key={c.id} value={c.id}>{c.criterion_number}. {c.criterion_text}</option>))}
        </select>
      </div>
      <Card className="border-dashed">
        <CardContent className="p-6 text-center">
          <input ref={fileInputRef} type="file" multiple accept="image/*,video/*,audio/*,.pdf,.docx,.xlsx" onChange={handleFileSelect} className="hidden" />
          <Upload className="mx-auto h-12 w-12 text-muted-foreground mb-4" />
          <p className="text-sm text-muted-foreground mb-2">Drag files here or click to browse</p>
          <p className="text-xs text-muted-foreground">JPG, PNG, WEBP, MP4, MOV, WEBM, MP3, WAV, PDF, DOCX, XLSX (Max 100MB)</p>
          <Button onClick={() => fileInputRef.current?.click()} disabled={uploading || !selectedCriterion} className="mt-4"><Upload className="mr-2 h-4 w-4" />Select Files</Button>
        </CardContent>
      </Card>
      {uploading && <div><p className="text-sm mb-1">Uploading... {progress.toFixed(0)}%</p><Progress value={progress} /></div>}
      {error && <p className="text-sm text-red-600">{error}</p>}
      {files.length > 0 && (
        <div className="space-y-2">
          <h3 className="font-medium">Uploaded ({files.length})</h3>
          {files.map((f) => (<div key={f.id} className="flex items-center gap-3 p-3 border rounded-lg">{getFileIcon(f.file_type)}<div className="flex-1 min-w-0"><p className="text-sm font-medium truncate">{f.system_filename}</p><p className="text-xs text-muted-foreground">{f.file_type.toUpperCase()} | {(f.file_size_bytes / 1024).toFixed(1)} KB</p></div><Badge variant="secondary">v{f.version}</Badge><CheckCircle className="h-4 w-4 text-green-500" /></div>))}
        </div>
      )}
    </div>
  );
}

import { useEffect, useState } from 'react';
import { evidenceApi } from '../../api/evidence';
import type { EvidenceFile } from '../../types/assessment';

export function EvidenceGallery({ sessionId }: { sessionId: string }) {
  const [files, setFiles] = useState<EvidenceFile[]>([]);
  useEffect(() => { evidenceApi.list(sessionId).then((r) => setFiles(r.data)).catch(() => {}); }, [sessionId]);
  if (files.length === 0) return <p className="text-muted-foreground text-sm">No evidence uploaded</p>;
  return (
    <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
      {files.map((f) => (<div key={f.id} className="border rounded p-3"><p className="text-sm font-medium truncate">{f.original_filename}</p><p className="text-xs text-muted-foreground">{f.file_type} • v{f.version}</p></div>))}
    </div>
  );
}

import { useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from '../ui/card';
import { Button } from '../ui/button';
import { Textarea } from '../ui/textarea';
import { verificationApi } from '../../api/verification';

export function VerificationPanel({ sessionId, onVerified }: { sessionId: string; onVerified: () => void }) {
  const [decision, setDecision] = useState('approved');
  const [comments, setComments] = useState('');
  const [reasons, setReasons] = useState('');

  const handleVerify = async () => {
    try {
      await verificationApi.verify(sessionId, { verifier_type: 'internal', decision, comments, rejection_reasons: reasons });
      onVerified();
    } catch (err: any) { alert(err.response?.data?.detail || 'Verification failed'); }
  };

  return (
    <Card>
      <CardHeader><CardTitle>Verification Decision</CardTitle></CardHeader>
      <CardContent className="space-y-3">
        <select value={decision} onChange={(e) => setDecision(e.target.value)} className="w-full p-2 border rounded">
          <option value="approved">Approve</option>
          <option value="returned_for_correction">Return for Correction</option>
          <option value="rejected">Reject</option>
        </select>
        <Textarea placeholder="Comments" value={comments} onChange={(e) => setComments(e.target.value)} rows={3} />
        {decision !== 'approved' && <Textarea placeholder="Rejection reasons" value={reasons} onChange={(e) => setReasons(e.target.value)} rows={3} />}
        <Button onClick={handleVerify} className="w-full">Submit Verification</Button>
      </CardContent>
    </Card>
  );
}
